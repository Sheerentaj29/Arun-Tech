                      </thead>
                      <tbody>
                        {twsRows.map((row, idx) => (
                          <tr key={idx} className="hover:bg-muted/30">
                            <td className="border border-zinc-300 px-2 py-1 text-center font-mono">{idx + 1}</td>

                            <td className="border border-zinc-300 px-2 py-1">{row.observationResult || "—"}</td>
                          </tr>
                        ))}
                      </tbody>
                    </table>
                  </div>
                </div>
              )}
            </div>
          );
        })()}
