     1→     1→// Server-side PDF generator for the Pipeline Thickness Measurement Report.
     2→     2→// Uses pdfkit (pure JS, no native deps).
     3→     3→//
     4→     4→// Fixes applied:
     5→     5→//   1. Dynamic row height in header info grid — long MOC values wrap
     6→     6→//      without overlapping adjacent fields.
     7→     7→//   2. Title/subtitle spacing — increased gap so no text overlaps.
     8→     8→//   3. AutoCAD diagrams — images whose caption or filename contains
     9→     9→//      "diagram", "isometric", "autocad", or "drawing" are treated as
    10→    10→//      diagrams. Each diagram gets its OWN FULL PAGE, displayed BEFORE
    11→    11→//      the measurement table. Regular photos stay in the gallery at the
    12→    12→//      end.
    13→    13→//   4. Dynamic page handling — 0 diagrams = table starts right after
    14→    14→//      header; N diagrams = N full pages, then table starts on a fresh
    15→    15→//      page.
    16→    16→
    17→    17→import PDFDocument from "pdfkit";
    18→    18→import type { Writable } from "stream";
    19→    19→import type { PipelineInspection, ThicknessMeasurement, Client, Plant, User, ReportPhoto } from "@prisma/client";
    20→    20→import { readFileSync, existsSync } from "fs";
    21→    21→
    22→    22→export interface PdfReportInput {
    23→    23→  report: PipelineInspection & {
    24→    24→    client: Pick<Client, "id" | "name">;
    25→    25→    plant: Pick<Plant, "id" | "name" | "location" | "address">;
    26→    26→    inspector: Pick<User, "id" | "name" | "email">;
    27→    27→    approver: Pick<User, "id" | "name" | "email"> | null;
    28→    28→    reportPhotos?: ReportPhoto[];
    29→    29→  };
    30→    30→  measurements: ThicknessMeasurement[];
    31→    31→  photos?: ReportPhoto[];
    32→    32→}
    33→    33→
    34→    34→type Doc = PDFDocument.PDFDocument;
    35→    35→
    36→    36→const PAGE_SIZES = {
    37→    37→  landscape: [841.89, 595.28] as [number, number],
    38→    38→  portrait: [595.28, 841.89] as [number, number],
    39→    39→};
    40→    40→
    41→    41→const MARGIN = 28;
    42→    42→const HEADER_BG = "#3a7d3a";
    43→    43→const HEADER_FG = "#ffffff";
    44→    44→
    45→    45→// Keywords that identify a photo as an AutoCAD diagram
    46→    46→const DIAGRAM_KEYWORDS = ["diagram", "isometric", "autocad", "drawing", "dwg", "iso"];
    47→    47→
    48→    48→function isDiagram(photo: ReportPhoto): boolean {
    49→    49→  const caption = (photo.caption ?? "").toLowerCase();
    50→    50→  const fileName = (photo.fileName ?? "").toLowerCase();
    51→    51→  return DIAGRAM_KEYWORDS.some((kw) => caption.includes(kw) || fileName.includes(kw));
    52→    52→}
    53→    53→
    54→    54→interface Column {
    55→    55→  key: string;
    56→    56→  header: string;
    57→    57→  width: number;
    58→    58→  vertical?: boolean;
    59→    59→}
    60→    60→
    61→    61→const SYSTEM_COLUMNS_LANDSCAPE: Column[] = [
    62→    62→  { key: "diaInch",      header: "Dia mm\n(inch)",        width: 50 },
    63→    63→  { key: "schedule",     header: "Pipe\nSchedule",       width: 55 },
    64→    64→  { key: "nominal",      header: "Nominal\nThickness\n(mm)", width: 55 },
    65→    65→  { key: "component",    header: "Component\nType",      width: 60 },
    66→    66→  { key: "pointNo",      header: "Point\nNo",             width: 40 },
    67→    67→  { key: "north",        header: "12 o'clock\nNorth\n(mm)", width: 55, vertical: true },
    68→    68→  { key: "east",         header: "3 o'clock\nEast\n(mm)",  width: 55, vertical: true },
    69→    69→  { key: "south",        header: "6 o'clock\nSouth\n(mm)", width: 55, vertical: true },
    70→    70→  { key: "west",         header: "9 o'clock\nWest\n(mm)",  width: 55, vertical: true },
    71→    71→  { key: "obsMin",       header: "Observed\nMinimum\n(mm)", width: 60, vertical: true },
    72→    72→  { key: "findings",     header: "Findings / Comments",   width: 0 },
    73→    73→];
    74→    74→
    75→    75→const SYSTEM_COLUMNS_PORTRAIT: Column[] = [
    76→    76→  { key: "diaInch",      header: "Dia\n(inch)",      width: 35 },
    77→    77→  { key: "schedule",     header: "Schedule",         width: 42 },
    78→    78→  { key: "nominal",      header: "Nominal\n(mm)",    width: 42 },
    79→    79→  { key: "component",    header: "Type",             width: 40 },
    80→    80→  { key: "pointNo",      header: "Pt\nNo",            width: 28 },
    81→    81→  { key: "north",        header: "N\n(mm)",           width: 38 },
    82→    82→  { key: "east",         header: "E\n(mm)",           width: 38 },
    83→    83→  { key: "south",        header: "S\n(mm)",           width: 38 },
    84→    84→  { key: "west",         header: "W\n(mm)",           width: 38 },
    85→    85→  { key: "obsMin",       header: "Min\n(mm)",         width: 42 },
    86→    86→  { key: "findings",     header: "Findings / Comments", width: 0 },
    87→    87→];
    88→    88→
    89→    89→function calcColumnWidths(columns: Column[], pageWidth: number): number[] {
    90→    90→  const usable = pageWidth - 2 * MARGIN;
    91→    91→  let fixed = 0;
    92→    92→  for (const c of columns) if (c.width > 0) fixed += c.width;
    93→    93→  const fillW = usable - fixed;
    94→    94→  return columns.map((c) => (c.width > 0 ? c.width : Math.max(100, fillW)));
    95→    95→}
    96→    96→
    97→    97→function getCustomColumns(report: any): Array<{ id: string; name: string; unit: string }> {
    98→    98→  try {
    99→    99→    const cc = JSON.parse(report.customColumns || "[]");
   100→   100→    if (Array.isArray(cc)) return cc;
   101→   101→  } catch {}
   102→   102→  return [];
   103→   103→}
   104→   104→
   105→   105→function getCustomColumnValue(measurement: any, colId: string): string {
   106→   106→  try {
   107→   107→    const vals = JSON.parse(measurement.customColumnValues || "{}");
   108→   108→    return vals[colId] ?? "";
   109→   109→  } catch { return ""; }
   110→   110→}
   111→   111→
   112→   112→function fmtMeasurement(v: number | null, status: string): string {
   113→   113→  if (status === "NotApplicable") return "N/A";
   114→   114→  if (status === "NotMeasured" || v === null || v === undefined) return "—";
   115→   115→  return v.toFixed(2);
   116→   116→}
   117→   117→
   118→   118→/**
   119→   119→ * Calculate the wrapped height of a text string at a given font size
   120→   120→ * within a given width. Used to determine dynamic row heights so long
   121→   121→ * values (like MOC) don't overlap.
   122→   122→ */
   123→   123→function textHeight(doc: Doc, text: string, fontSize: number, maxWidth: number): number {
   124→   124→  doc.fontSize(fontSize).font("Helvetica");
   125→   125→  return doc.heightOfString(text ?? "—", { width: maxWidth });
   126→   126→}
   127→   127→
   128→   128→export async function generatePipelineReportPdf(
   129→   129→  input: PdfReportInput,
   130→   130→  out: Writable,
   131→   131→  orientation: "landscape" | "portrait" = "landscape",
   132→   132→): Promise<void> {
   133→   133→  const pageSize = PAGE_SIZES[orientation];
   134→   134→  const pageW = pageSize[0];
   135→   135→  const pageH = pageSize[1];
   136→   136→
   137→   137→  const doc = new PDFDocument({
   138→   138→    size: pageSize,
   139→   139→    margins: { top: MARGIN, bottom: MARGIN, left: MARGIN, right: MARGIN },
   140→   140→    layout: orientation,
   141→   141→    info: {
   142→   142→      Title: `Pipeline Thickness Measurement Report ${input.report.reportNumber}`,
   143→   143→      Author: input.report.inspectorName,
   144→   144→      Subject: input.report.pipelineDescription ?? "UT Thickness Measurement",
   145→   145→      Producer: "Inspection Report Automation System",
   146→   146→    },
   147→   147→  });
   148→   148→  doc.pipe(out);
   149→   149→
   150→   150→  const report = input.report;
   151→   151→  const measurements = input.measurements;
   152→   152→  const allPhotos = input.photos ?? input.report.reportPhotos ?? [];
   153→   153→  const customCols = getCustomColumns(report);
   154→   154→
   155→   155→  // ── LINKED ENGINEERING DRAWINGS ──
   156→   156→  // These are drawings linked to the report via the DrawingInspectionLink
   157→   157→  // model. Each gets its own FULL PAGE before the measurement table.
   158→   158→  // The PDF route loads them with their active revision's file path.
   159→   159→  const linkedDrawings: any[] = (report as any).linkedDrawings ?? [];
   160→   160→
   161→   161→  // Regular inspection photos stay in the gallery at the end.
   162→   162→  // (We no longer filter photos by "diagram" keywords — drawings come
   163→   163→  // from the dedicated Engineering Drawings module.)
   164→   164→  const regularPhotos = allPhotos;
   165→   165→
   166→   166→  // Determine columns: system + custom
   167→   167→  const systemCols = orientation === "landscape" ? SYSTEM_COLUMNS_LANDSCAPE : SYSTEM_COLUMNS_PORTRAIT;
   168→   168→  const allCols: Column[] = [
   169→   169→    ...systemCols,
   170→   170→    ...customCols.map((cc) => ({
   171→   171→      key: `custom_${cc.id}`,
   172→   172→      header: cc.name + (cc.unit ? `\n(${cc.unit})` : ""),
   173→   173→      width: orientation === "landscape" ? 60 : 40,
   174→   174→    })),
   175→   175→  ];
   176→   176→  const colWidths = calcColumnWidths(allCols, pageW);
   177→   177→
   178→   178→  let pageNo = 0;
   179→   179→  const drawHeaderAndFooter = () => {
   180→   180→    pageNo += 1;
   181→   181→    doc.fontSize(7).fillColor("#666");
   182→   182→    doc.text(
   183→   183→      `Page ${pageNo} | Generated ${new Date().toISOString().slice(0, 19).replace("T", " ")} | Inspection Report Automation System`,
   184→   184→      MARGIN, pageH - 18,
   185→   185→      { width: pageW - 2 * MARGIN, align: "center" },
   186→   186→    );
   187→   187→    doc.fillColor("#000");
   188→   188→  };
   189→   189→
   190→   190→  const addNewPage = () => {
   191→   191→    doc.addPage({
   192→   192→      size: pageSize,
   193→   193→      layout: orientation,
   194→   194→      margins: { top: MARGIN, bottom: MARGIN, left: MARGIN, right: MARGIN },
   195→   195→    });
   196→   196→    drawHeaderAndFooter();
   197→   197→  };
   198→   198→
   199→   199→  drawHeaderAndFooter();
   200→   200→
   201→   201→  // ═══════════════════════════════════════════════════════════════
   202→   202→  // PAGE 1: TITLE + LOGOS + HEADER INFO GRID
   203→   203→  // ═══════════════════════════════════════════════════════════════
   204→   204→
   205→   205→  // ── FIX 2: Title with proper spacing ──
   206→   206→  const titleFontSize = orientation === "landscape" ? 13 : 11;
   207→   207→  const subtitleFontSize = orientation === "landscape" ? 8 : 7;
   208→   208→  const titleY = MARGIN;
   209→   209→
   210→   210→  // Draw the title
   211→   211→  doc.fontSize(titleFontSize).font("Helvetica-Bold");
   212→   212→  doc.text("PIPELINE THICKNESS MEASUREMENT REPORT", MARGIN, titleY, {
   213→   213→    width: pageW - 2 * MARGIN, align: "center",
   214→   214→  });
   215→   215→
   216→   216→  // Calculate actual title height (handles wrapping)
   217→   217→  const titleHeight = doc.heightOfString("PIPELINE THICKNESS MEASUREMENT REPORT", {
   218→   218→    width: pageW - 2 * MARGIN, align: "center",
   219→   219→  });
   220→   220→
   221→   221→  // Draw subtitle with enough gap below the title (at least 6pt gap)
   222→   222→  const subtitleY = titleY + titleHeight + 6;
   223→   223→  doc.font("Helvetica").fontSize(subtitleFontSize);
   224→   224→  doc.text(
   225→   225→    `Report No: ${report.reportNumber}    |    Rev: ${report.revisionNumber}    |    Date: ${new Date(report.inspectionDate).toISOString().slice(0, 10)}    |    Status: ${report.status}`,
   226→   226→    MARGIN, subtitleY,
   227→   227→    { width: pageW - 2 * MARGIN, align: "center" },
   228→   228→  );
   229→   229→
   230→   230→  const subtitleHeight = doc.heightOfString(
   231→   231→    `Report No: ${report.reportNumber}    |    Rev: ${report.revisionNumber}    |    Date: ${new Date(report.inspectionDate).toISOString().slice(0, 10)}    |    Status: ${report.status}`,
   232→   232→    { width: pageW - 2 * MARGIN, align: "center" },
   233→   233→  );
   234→   234→
   235→   235→  // ── LOGO PLACEHOLDERS (positioned below subtitle with gap) ──
   236→   236→  const logoY = subtitleY + subtitleHeight + 6;
   237→   237→  const logoH = 36;
   238→   238→  doc.rect(MARGIN, logoY, 110, logoH).stroke();
   239→   239→  doc.fontSize(7).fillColor("#666");
   240→   240→  doc.text("SERVICE COMPANY\nLOGO", MARGIN, logoY + 10, { width: 110, align: "center" });
   241→   241→  doc.rect(pageW - MARGIN - 110, logoY, 110, logoH).stroke();
   242→   242→  const clientShort = report.client.name.length > 28 ? report.client.name.slice(0, 28) : report.client.name;
   243→   243→  doc.text(`${clientShort}\nCLIENT LOGO`, pageW - MARGIN - 110, logoY + 10, { width: 110, align: "center" });
   244→   244→  doc.fillColor("#000");
   245→   245→
   246→   246→  // ── FIX 1: HEADER INFO GRID with dynamic row heights ──
   247→   247→  // Long values (like MOC) wrap to multiple lines. Calculate each row's
   248→   248→  // height dynamically so text never overlaps.
   249→   249→  const infoY = logoY + logoH + 10;
   250→   250→  const labelW = orientation === "landscape" ? 150 : 110;
   251→   251→  const valueW = (pageW - 2 * MARGIN - 2 * labelW) / 2;
   252→   252→  const minRowH = 14; // minimum height per row
   253→   253→  const fontSizeInfo = orientation === "landscape" ? 8 : 7;
   254→   254→
   255→   255→  const infoRows: Array<[string, string | null | undefined]> = [
   256→   256→    ["Client", report.client.name],
   257→   257→    ["Location", report.plant.location],
   258→   258→    ["Plant Name", report.plant.name],
   259→   259→    ["Inspection Area", report.inspectionArea],
   260→   260→    ["Pipeline Description", report.pipelineDescription],
   261→   261→    ["Pipeline Identification No.", report.pipelineIdentificationNumber],
   262→   262→    ["Isometric Drawing No.", report.isometricDrawingNumber],
   263→   263→    ["Material of Construction (MOC)", report.materialOfConstruction],
   264→   264→    ["Applicable Standard", `${report.applicableStandard ?? "—"} ${report.standardEdition ?? ""}`.trim()],
   265→   265→    ["Inspector", report.inspectorName],
   266→   266→  ];
   267→   267→
   268→   268→  // Group rows into pairs (left + right columns) and calculate the max
   269→   269→  // height needed for each pair.
   270→   270→  const rowPairs: Array<{ left: [string, string]; right: [string, string]; height: number }> = [];
   271→   271→  for (let i = 0; i < infoRows.length; i += 2) {
   272→   272→    const left = infoRows[i];
   273→   273→    const right = infoRows[i + 1] ?? ["", "—"];
   274→   274→
   275→   275→    // Calculate the wrapped height for each value in this pair
   276→   276→    const leftLabelH = textHeight(doc, left[0], fontSizeInfo, labelW - 8);
   277→   277→    const leftValStr = left[1] ?? "—";
   278→   278→    const leftValH = textHeight(doc, leftValStr, fontSizeInfo, valueW - 8);
   279→   279→    const rightLabelH = textHeight(doc, right[0], fontSizeInfo, labelW - 8);
   280→   280→    const rightValStr = right[1] ?? "—";
   281→   281→    const rightValH = textHeight(doc, rightValStr, fontSizeInfo, valueW - 8);
   282→   282→
   283→   283→    // The row height is the maximum of all four heights, with padding
   284→   284→    const maxContentH = Math.max(leftLabelH, leftValH, rightLabelH, rightValH);
   285→   285→    const pairHeight = Math.max(minRowH, maxContentH + 4); // +4 for padding
   286→   286→
   287→   287→    rowPairs.push({
   288→   288→      left: [left[0], leftValStr],
   289→   289→      right: [right[0], rightValStr],
   290→   290→      height: pairHeight,
   291→   291→    });
   292→   292→  }
   293→   293→
   294→   294→  // Draw each pair with its calculated height
   295→   295→  let currentY = infoY;
   296→   296→  for (const pair of rowPairs) {
   297→   297→    const h = pair.height;
   298→   298→    // Left label
   299→   299→    doc.font("Helvetica-Bold").fontSize(fontSizeInfo);
   300→   300→    doc.rect(MARGIN, currentY, labelW, h).stroke();
   301→   301→    doc.text(pair.left[0], MARGIN + 4, currentY + 2, { width: labelW - 8, align: "left", height: h });
   302→   302→    // Left value — use height option to clip text within the cell
   303→   303→    doc.font("Helvetica").fontSize(fontSizeInfo);
   304→   304→    doc.rect(MARGIN + labelW, currentY, valueW, h).stroke();
   305→   305→    doc.text(pair.left[1], MARGIN + labelW + 4, currentY + 2, { width: valueW - 8, align: "left", height: h });
   306→   306→
   307→   307→    // Right label
   308→   308→    const rightX = MARGIN + labelW + valueW;
   309→   309→    doc.font("Helvetica-Bold").fontSize(fontSizeInfo);
   310→   310→    doc.rect(rightX, currentY, labelW, h).stroke();
   311→   311→    doc.text(pair.right[0], rightX + 4, currentY + 2, { width: labelW - 8, align: "left", height: h });
   312→   312→    // Right value
   313→   313→    doc.font("Helvetica").fontSize(fontSizeInfo);
   314→   314→    doc.rect(rightX + labelW, currentY, valueW, h).stroke();
   315→   315→    doc.text(pair.right[1], rightX + labelW + 4, currentY + 2, { width: valueW - 8, align: "left", height: h });
   316→   316→
   317→   317→    currentY += h;
   318→   318→  }
   319→   319→
   320→   320→  // ═══════════════════════════════════════════════════════════════
   321→   321→  // ENGINEERING DRAWINGS — each on its OWN FULL PAGE, before the table
   322→   322→  // ═══════════════════════════════════════════════════════════════
   323→   323→  if (linkedDrawings.length > 0) {
   324→   324→    for (let i = 0; i < linkedDrawings.length; i++) {
   325→   325→      const drawing = linkedDrawings[i];
   326→   326→      // Each drawing gets a fresh full page
   327→   327→      addNewPage();
   328→   328→
   329→   329→      const availW = pageW - 2 * MARGIN;
   330→   330→      const availH = pageH - 2 * MARGIN - 40; // room for header + caption
   331→   331→
   332→   332→      // Drawing header: title, number, revision
   333→   333→      doc.font("Helvetica-Bold").fontSize(11).fillColor("#000");
   334→   334→      doc.text(`${drawing.title}`, MARGIN, MARGIN, { width: availW, align: "center" });
   335→   335→      doc.font("Helvetica").fontSize(8).fillColor("#444");
   336→   336→      const subLine = `Drawing No: ${drawing.drawingNumber}  |  Revision: ${drawing.revisionNumber}  |  Type: ${drawing.drawingType}`;
   337→   337→      doc.text(subLine, MARGIN, MARGIN + 16, { width: availW, align: "center" });
   338→   338→
   339→   339→      // Client/plant info
   340→   340→      if (drawing.clientName || drawing.plantName) {
   341→   341→        const locLine = [drawing.clientName, drawing.plantName].filter(Boolean).join(" — ");
   342→   342→        doc.fontSize(7).fillColor("#666");
   343→   343→        doc.text(locLine, MARGIN, MARGIN + 30, { width: availW, align: "center" });
   344→   344→      }
   345→   345→      doc.fillColor("#000");
   346→   346→
   347→   347→      // Embed the drawing file
   348→   348→      const drawingY = MARGIN + 40;
   349→   349→      const drawingH = availH - 40;
   350→   350→
   351→   351→      if (drawing.filePath && existsSync(drawing.filePath)) {
   352→   352→        try {
   353→   353→          const imgBuf = readFileSync(drawing.filePath);
   354→   354→          const ext = (drawing.fileExtension || "").toLowerCase();
   355→   355→
   356→   356→          if (ext === ".pdf") {
   357→   357→            // For PDF drawings, we can't embed a PDF inside a PDF.
   358→   358→            // Show a placeholder with download instructions.
   359→   359→            doc.fillColor("#999").fontSize(10);
   360→   360→            doc.rect(MARGIN, drawingY, availW, drawingH).stroke("#ccc");
   361→   361→            doc.text(
   362→   362→              `[PDF Drawing: ${drawing.fileName}]\n\nThis drawing is a PDF file and cannot be embedded directly.\nDownload the original file to view it.`,
   363→   363→              MARGIN + 20, drawingY + drawingH / 2 - 20,
   364→   364→              { width: availW - 40, align: "center" },
   365→   365→            );
   366→   366→            doc.fillColor("#000");
   367→   367→          } else {
   368→   368→            // Image formats (PNG, JPEG, TIFF) — embed directly
   369→   369→            doc.image(imgBuf, MARGIN, drawingY, {
   370→   370→              fit: [availW, drawingH],
   371→   371→              align: "center",
   372→   372→              valign: "center",
   373→   373→            });
   374→   374→          }
   375→   375→        } catch (err) {
   376→   376→          // For DWG/DXF or unreadable files, show a clear message
   377→   377→          doc.fillColor("#999").fontSize(9);
   378→   378→          doc.rect(MARGIN, drawingY, availW, drawingH).stroke("#ccc");
   379→   379→          const ext = (drawing.fileExtension || "").toUpperCase();
   380→   380→          doc.text(
   381→   381→            `[${ext} File: ${drawing.fileName}]\n\nPreview not available for ${ext} files.\nDownload the original CAD file to view it in AutoCAD or compatible software.`,
   382→   382→            MARGIN + 20, drawingY + drawingH / 2 - 20,
   383→   383→            { width: availW - 40, align: "center" },
   384→   384→          );
   385→   385→          doc.fillColor("#000");
   386→   386→        }
   387→   387→      } else {
   388→   388→        doc.fillColor("#999").fontSize(9);
   389→   389→        doc.rect(MARGIN, drawingY, availW, drawingH).stroke("#ccc");
   390→   390→        doc.text(
   391→   391→          `[File missing: ${drawing.fileName}]\n\nThe drawing file could not be found on disk.`,
   392→   392→          MARGIN + 20, drawingY + drawingH / 2 - 15,
   393→   393→          { width: availW - 40, align: "center" },
   394→   394→        );
   395→   395→        doc.fillColor("#000");
   396→   396→      }
   397→   397→    }
   398→   398→  }

   401→   401→  // MEASUREMENT TABLE — starts on a FRESH PAGE (after all diagrams)
   402→   402→  // ═══════════════════════════════════════════════════════════════
   403→   403→
   404→   404→  // If there are diagrams OR the header info grid filled most of page 1,
   405→   405→  // start the table on a fresh page. Otherwise, continue on the same page
   406→   406→  // if there's room.
   407→   407→  const tableHeaderH = 36;
   408→   408→  const rowH2 = orientation === "landscape" ? 28 : 22;
   409→   409→  const maxRowY = pageH - MARGIN - 20; // leave room for footer
   410→   410→
   411→   411→  // If there are linked drawings, the table MUST start on a fresh page
   412→   412→  let y: number;
   413→   413→
   414→   414→  if (linkedDrawings.length > 0) {
   415→   415→    addNewPage();
   416→   416→    drawTableHeader(doc, allCols, colWidths, MARGIN, pageW);
   417→   417→    y = MARGIN + tableHeaderH;
   418→   418→  } else {
   419→   419→    if (currentY + tableHeaderH + rowH2 > maxRowY) {
   420→   420→      addNewPage();
   421→   421→      drawTableHeader(doc, allCols, colWidths, MARGIN, pageW);
   422→   422→      y = MARGIN + tableHeaderH;
   423→   423→    } else {
   424→   424→      currentY += 10;
   425→   425→      drawTableHeader(doc, allCols, colWidths, currentY, pageW);
   426→   426→      y = currentY + tableHeaderH;
   427→   427→    }
   428→   428→  }
   429→   429→
   430→   430→  // ── TABLE ROWS ──
   431→   431→  for (const m of measurements) {
   432→   432→    if (y + rowH2 > maxRowY) {
   433→   433→      addNewPage();
   434→   434→      drawTableHeader(doc, allCols, colWidths, MARGIN, pageW);
   435→   435→      y = MARGIN + tableHeaderH;
   436→   436→    }
   437→   437→    drawMeasurementRow(doc, m, allCols, colWidths, y, report, customCols);
   438→   438→    y += rowH2;
   439→   439→  }
   440→   440→
   441→   441→  // ── INSPECTION DESCRIPTION ──
   442→   442→  if (report.inspectionDescription) {
   443→   443→    if (y + 60 > maxRowY) {
   444→   444→      addNewPage();
   445→   445→      y = MARGIN;
   446→   446→    }
   447→   447→    y += 10;
   448→   448→    doc.font("Helvetica-Bold").fontSize(9).fillColor("#000");
   449→   449→    doc.text("Inspection Description", MARGIN, y);
   450→   450→    y += 14;
   451→   451→    doc.font("Helvetica").fontSize(8);
   452→   452→    const descHeight = doc.heightOfString(report.inspectionDescription, {
   453→   453→      width: pageW - 2 * MARGIN,
   454→   454→    });
   455→   455→    doc.text(report.inspectionDescription, MARGIN, y, {
   456→   456→      width: pageW - 2 * MARGIN, align: "left",
   457→   457→    });
   458→   458→    y += descHeight + 6;
   459→   459→  }
   460→   460→
   461→   461→  // ── FINDINGS SUMMARY ──
   462→   462→  if (report.findingsSummary) {
   463→   463→    if (y + 40 > maxRowY) {
   464→   464→      addNewPage();
   465→   465→      y = MARGIN;
   466→   466→    }
   467→   467→    y += 8;
   468→   468→    doc.font("Helvetica-Bold").fontSize(9);
   469→   469→    doc.text("Findings Summary", MARGIN, y);
   470→   470→    y += 14;
   471→   471→    doc.font("Helvetica").fontSize(8);
   472→   472→    const findingsH = doc.heightOfString(report.findingsSummary ?? "—", {
   473→   473→      width: pageW - 2 * MARGIN,
   474→   474→    });
   475→   475→    doc.text(report.findingsSummary ?? "—", MARGIN, y, {
   476→   476→      width: pageW - 2 * MARGIN, align: "left",
   477→   477→    });
   478→   478→    y += findingsH + 6;
   479→   479→  }
   480→   480→
   481→   481→  // ── COMMENTS ──
   482→   482→  if (report.comments) {
   483→   483→    if (y + 30 > maxRowY) {
   484→   484→      addNewPage();
   485→   485→      y = MARGIN;
   486→   486→    }
   487→   487→    y += 6;
   488→   488→    doc.font("Helvetica-Bold").fontSize(9);
   489→   489→    doc.text("Comments", MARGIN, y);
   490→   490→    y += 14;
   491→   491→    doc.font("Helvetica").fontSize(8);
   492→   492→    const commentsH = doc.heightOfString(report.comments ?? "—", {
   493→   493→      width: pageW - 2 * MARGIN,
   494→   494→    });
   495→   495→    doc.text(report.comments ?? "—", MARGIN, y, {
   496→   496→      width: pageW - 2 * MARGIN, align: "left",
   497→   497→    });
   498→   498→    y += commentsH + 6;
   499→   499→  }
   500→   500→
   501→   501→  // ── REGULAR PHOTOS (gallery — NOT diagrams) ──
   502→   502→  if (regularPhotos.length > 0) {
   503→   503→    if (y + 100 > maxRowY) {
   504→   504→      addNewPage();
   505→   505→      y = MARGIN;
   506→   506→    }
   507→   507→    y += 10;
   508→   508→    doc.font("Helvetica-Bold").fontSize(10).fillColor("#000");
   509→   509→    doc.text("Photographs", MARGIN, y);
   510→   510→    y += 20;
   511→   511→
   512→   512→    const photoW = orientation === "landscape" ? 180 : 140;
   513→   513→    const photoH = orientation === "landscape" ? 120 : 95;
   514→   514→    const gap = 10;
   515→   515→    let photosPerRow = Math.floor((pageW - 2 * MARGIN + gap) / (photoW + gap));
   516→   516→    if (photosPerRow < 1) photosPerRow = 1;
   517→   517→
   518→   518→    for (let i = 0; i < regularPhotos.length; i++) {
   519→   519→      const photo = regularPhotos[i];
   520→   520→      const col = i % photosPerRow;
   521→   521→      const row = Math.floor(i / photosPerRow);
   522→   522→      const px = MARGIN + col * (photoW + gap);
   523→   523→      const py = y + row * (photoH + 30);
   524→   524→
   525→   525→      if (py + photoH + 20 > maxRowY) {
   526→   526→        addNewPage();
   527→   527→        y = MARGIN + 20;
   528→   528→      }
   529→   529→
   530→   530→      doc.rect(px, py, photoW, photoH).stroke("#999");
   531→   531→
   532→   532→      if (photo.filePath && existsSync(photo.filePath)) {
   533→   533→        try {
   534→   534→          const imgBuf = readFileSync(photo.filePath);
   535→   535→          doc.image(imgBuf, px + 1, py + 1, {
   536→   536→            fit: [photoW - 2, photoH - 2],
   537→   537→            align: "center",
   538→   538→            valign: "center",
   539→   539→          });
   540→   540→        } catch (err) {
   541→   541→          doc.fillColor("#999").fontSize(7);
   542→   542→          doc.text("[Image unreadable]", px + 4, py + photoH / 2 - 5, {
   543→   543→            width: photoW - 8, align: "center",
   544→   544→          });
   545→   545→          doc.fillColor("#000");
   546→   546→        }
   547→   547→      } else {
   548→   548→        doc.fillColor("#999").fontSize(7);
   549→   549→        doc.text(`[File missing: ${photo.fileName}]`, px + 4, py + photoH / 2 - 5, {
   550→   550→          width: photoW - 8, align: "center",
   551→   551→        });
   552→   552→        doc.fillColor("#000");
   553→   553→      }
   554→   554→
   555→   555→      doc.font("Helvetica").fontSize(7).fillColor("#333");
   556→   556→      const caption = photo.caption || photo.fileName;
   557→   557→      doc.text(caption, px, py + photoH + 2, { width: photoW, align: "center" });
   558→   558→      doc.fillColor("#000");
   559→   559→    }
   560→   560→    y += Math.ceil(regularPhotos.length / photosPerRow) * (photoH + 30);
   561→   561→  }
   562→   562→
   563→   563→  // ── SIGN-OFF ──
   564→   564→  if (y + 80 > maxRowY) {
   565→   565→    addNewPage();
   566→   566→    y = MARGIN;
   567→   567→  }
   568→   568→  y += 10;
   569→   569→  doc.font("Helvetica-Bold").fontSize(9).fillColor("#000");
   570→   570→  doc.text("Sign-off", MARGIN, y);
   571→   571→  y += 18;
   572→   572→
   573→   573→  const signW = (pageW - 2 * MARGIN - 2 * 8) / 3;
   574→   574→  const signBlocks = [
   575→   575→    { label: "Inspected by", name: report.inspectorName, dateVal: new Date(report.inspectionDate).toISOString().slice(0, 10) },
   576→   576→    { label: "Reviewed / Approved by", name: report.approver?.name ?? "—", dateVal: report.approvedAt?.slice(0, 10) ?? "—" },
   577→   577→    { label: "Revision", name: `Rev ${report.revisionNumber}`, dateVal: report.status },
   578→   578→  ];
   579→   579→  for (let i = 0; i < signBlocks.length; i++) {
   580→   580→    const b = signBlocks[i];
   581→   581→    const x = MARGIN + i * (signW + 8);
   582→   582→    doc.font("Helvetica").fontSize(7).fillColor("#666");
   583→   583→    doc.rect(x, y, signW, 50).stroke("#999");
   584→   584→    doc.text(b.label, x + 4, y + 4, { width: signW - 8 });
   585→   585→    doc.fillColor("#000");
   586→   586→    doc.font("Helvetica-Bold").fontSize(8);
   587→   587→    doc.text(b.name, x + 4, y + 16, { width: signW - 8 });
   588→   588→    doc.font("Helvetica").fontSize(7).fillColor("#666");
   589→   589→    doc.text(`Date / Status: ${b.dateVal}`, x + 4, y + 36, { width: signW - 8 });
   590→   590→    doc.fillColor("#000");
   591→   591→  }
   592→   592→
   593→   593→  // ── OVERRIDE NOTE ──
   594→   594→  if (report.overrideReason) {
   595→   595→    y += 60;
   596→   596→    doc.font("Helvetica-Bold").fontSize(8).fillColor("#a00");
   597→   597→    doc.text("OVERRIDE APPLIED (recorded):", MARGIN, y, { width: 200 });
   598→   598→    doc.font("Helvetica").fillColor("#000").fontSize(7);
   599→   599→    doc.text(report.overrideReason, MARGIN + 140, y, { width: pageW - 2 * MARGIN - 140 });
   600→   600→  }
   601→   601→
   602→   602→  doc.end();
   603→   603→}

   607→   607→function drawTableHeader(doc: Doc, columns: Column[], colWidths: number[], y: number, pageW: number) {
   608→   608→  const usable = pageW - 2 * MARGIN;
   609→   609→  doc.rect(MARGIN, y, usable, 36).fill(HEADER_BG).fillColor(HEADER_FG);
   610→   610→  let x = MARGIN;
   611→   611→  doc.font("Helvetica-Bold").fontSize(7).fillColor(HEADER_FG);
   612→   612→  for (let i = 0; i < columns.length; i++) {
   613→   613→    const c = columns[i];
   614→   614→    const w = colWidths[i];
   615→   615→    if (c.vertical) {
   616→   616→      const lines = c.header.split("\n");
   617→   617→      const joined = lines.join(" ");
   618→   618→      doc.save();
   619→   619→      doc.translate(x + w / 2 + 3, y + 32);
   620→   620→      doc.rotate(-90);
   621→   621→      doc.text(joined, 0, 0, { width: 30, align: "left" });
   622→   622→      doc.restore();
   623→   623→    } else {
   624→   624→      const lines = c.header.split("\n");
   625→   625→      let ly = y + 4;
   626→   626→      for (const ln of lines) {
   627→   627→        doc.text(ln, x + 3, ly, { width: w - 6, align: "center" });
   628→   628→        ly += 9;
   629→   629→      }
   630→   630→    }
   631→   631→    doc.rect(x, y, w, 36).stroke("#1b4d1b");
   632→   632→    x += w;
   633→   633→  }
   634→   634→  doc.fillColor("#000");
   635→   635→}

   637→   637→function drawMeasurementRow(
   638→   638→  doc: Doc,
   639→   639→  m: ThicknessMeasurement,
   640→   640→  columns: Column[],
   641→   641→  colWidths: number[],
   642→   642→  y: number,
   643→   643→  report: any,
   644→   644→  customCols: any[],
   645→   645→) {
   646→   646→  const rowH = 28;
   647→   647→  const usable = doc.page.width - 2 * MARGIN;
   648→   648→  doc.rect(MARGIN, y, usable, rowH).stroke("#999");
   649→   649→  let x = MARGIN;
   650→   650→  const cellPad = 3;
   651→   651→  doc.font("Helvetica").fontSize(8).fillColor("#000");
   652→   652→
   653→   653→  for (let i = 0; i < columns.length; i++) {
   654→   654→    const c = columns[i];
   655→   655→    const w = colWidths[i];
   656→   656→    let text = "";
   657→   657→    switch (c.key) {
   658→   658→      case "diaInch": text = m.pipeSizeInch; break;
   659→   659→      case "schedule": text = m.pipeSchedule; break;
   660→   660→      case "nominal": text = m.nominalThicknessMm !== null ? m.nominalThicknessMm.toFixed(2) : "—"; break;
   661→   661→      case "component": text = m.componentTypeCode; break;
   662→   662→      case "pointNo": text = m.pointNo; break;
   663→   663→      case "north": text = fmtMeasurement(m.northMm, m.northStatus); break;
   664→   664→      case "east": text = fmtMeasurement(m.eastMm, m.eastStatus); break;
   665→   665→      case "south": text = fmtMeasurement(m.southMm, m.southStatus); break;
   666→   666→      case "west": text = fmtMeasurement(m.westMm, m.westStatus); break;
   667→   667→      case "obsMin": text = m.observedMinimumMm !== null ? m.observedMinimumMm.toFixed(2) : "—"; break;
   668→   668→      case "findings": text = m.findingsComments ?? ""; break;
   669→   669→      default:
   670→   670→        if (c.key.startsWith("custom_")) {
   671→   671→          const colId = c.key.replace("custom_", "");
   672→   672→          text = getCustomColumnValue(m, colId);
   673→   673→        }
   674→   674→        break;
   675→   675→    }
   676→   676→    doc.rect(x, y, w, rowH).stroke("#bbb");
   677→   677→    const maxW = w - 2 * cellPad;
   678→   678→    const wrapped = doc.widthOfString(text) > maxW
   679→   679→      ? text.slice(0, Math.floor((maxW / Math.max(1, doc.widthOfString(text))) * text.length)) + "…"
   680→   680→      : text;
   681→   681→    doc.text(wrapped, x + cellPad, y + 9, { width: maxW, align: c.key === "findings" ? "left" : "center" });
   682→   682→    x += w;
   683→   683→  }
   684→   684→}
