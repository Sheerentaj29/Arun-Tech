   153→   153→// PIPE STANDARD REFERENCE DATA (ASME B36.10M / B36.19M)
   154→   154→// =====================================================
   155→   155→// Wall thickness values seeded from the publicly published ASME
   156→   156→// reference tables for the most commonly used nominal pipe sizes and
   157→   157→// schedules. These values MUST be re-verified against the licensed /
   158→   158→// official ASME standard edition by an authorized administrator before
   159→   159→// production use. An admin can edit, add, or delete entries, and every
   160→   160→// edit is recorded with verifiedBy + verifiedAt + audit log entry.
   161→   161→
   162→   162→model PipeStandard {
   163→   163→  id              String   @id @default(cuid())
   164→   164→  standard        String   // "ASME B36.10M" | "ASME B36.19M"
   165→   165→  edition         String   // e.g. "2018"
   166→   166→  nominalSizeInch String   // e.g. "2" or "1/2"
   167→   167→  nominalSizeMm   Float   // e.g. 60.3
   168→   168→  dn              String   // e.g. "DN50"
   169→   169→  schedule        String   // e.g. "SCH 40", "STD", "XS", "XXS", "SCH 40S"
   170→   170→  wallThicknessMm Float   // nominal wall thickness in mm
   171→   171→  oD_mm           Float?   // outside diameter (mm) when applicable
   172→   172→  isVerified      Boolean  @default(false)
   173→   173→  verifiedById    String?
   174→   174→  verifiedAt      DateTime?
   175→   175→  notes           String?
   176→   176→  createdAt       DateTime @default(now())
   177→   177→  updatedAt       DateTime @updatedAt
   178→   178→
   179→   179→  verifiedBy      User?    @relation("PipeStandardVerifiedBy", fields: [verifiedById], references: [id], onDelete: SetNull)
   180→   180→  measurements    ThicknessMeasurement[]
   181→   181→
   182→   182→  @@unique([standard, edition, nominalSizeInch, schedule])
   183→   183→  @@index([standard, nominalSizeInch])
   184→   184→}

   450→   450→// THICKNESS MEASUREMENT ROW
   451→   451→// =====================================================
   452→   452→// One row per measurement point. The first two columns (Dia mm (inch),
   453→   453→// Pipe Schedule) are stored as structured values; the third column
   454→   454→// (Nominal Thickness) is calculated from PipeStandard and may be
   455→   455→// overridden only with a recorded reason. The four clock-position
   456→   456→// measurements support an explicit status (Measured | NotMeasured |
   457→   457→// NotApplicable) so missing data is never converted to zero.

   466→   466→  // When a Thickness report contains multiple tables, each row belongs to
   467→   467→  // exactly one table group. Existing rows (and any new row in a
   468→   468→  // single-table report) get "default" — so the existing single-table
   469→   469→  // workflow keeps working unchanged. The list of table groups itself is
   470→   470→  // stored as JSON on the parent PipelineInspection.thicknessTablesConfig.
   471→   471→  tableGroup            String   @default("default")
   472→   472→
   473→   473→  // First two fixed columns
   474→   474→  pipeSizeInch          String   // e.g. "2", "1/2", "6" (matches PipeStandard.nominalSizeInch)
   475→   475→  pipeSchedule          String   // e.g. "SCH 40", "STD"
   476→   476→
   477→   477→  // Third column (calculated from standards; override captured separately)
   478→   478→  nominalThicknessMm    Float?   // snapshot from PipeStandard at row creation/update
   479→   479→  nominalOverride       Float?   // authorized override value (with reason at report level)
   480→   480→  standardId            String?  // FK to PipeStandard used to compute nominalThicknessMm
   481→   481→
   482→   482→  // Fourth column
   483→   483→  componentTypeId        String?
   484→   484→  componentTypeCode      String   // snapshot code (PIPE | ELBOW | TEE | REDUCER | FLANGE | OTHER)

   486→   486→  // Columns 5-9: clock-position measurements with explicit status
   487→   487→  northMm               Float?
   488→   488→  northStatus            String   @default("NotMeasured") // Measured | NotMeasured | NotApplicable
   489→   489→  eastMm                 Float?
   490→   490→  eastStatus             String   @default("NotMeasured")
   491→   491→  southMm               Float?
   492→   492→  southStatus            String   @default("NotMeasured")
   493→   493→  westMm                 Float?
   494→   494→  westStatus             String   @default("NotMeasured")
   495→   495→
   496→   496→  // Column 10: observed minimum (calculated; null if no valid measurements)
   497→   497→  observedMinimumMm     Float?
   498→   498→  observedMinimumOverride Float? // authorized override with reason
   499→   499→
   500→   500→  // Column 11: findings / comments
   501→   501→  findingsComments      String?
   502→   502→  // Custom column values stored as JSON: {"col_id_1": "5.2", "col_id_2": "N/A"}
   503→   503→  // Each key corresponds to a custom column defined on the parent PipelineInspection.
   504→   504→  customColumnValues    String?
   505→   505→
   506→   506→  createdAt             DateTime @default(now())
   507→   507→  updatedAt             DateTime @updatedAt
   508→   508→
   509→   509→  pipelineInspection    PipelineInspection @relation(fields: [pipelineInspectionId], references: [id], onDelete: Cascade)
   510→   510→  componentType         ComponentType?     @relation(fields: [componentTypeId], references: [id], onDelete: SetNull)
   511→   511→  standard              PipeStandard?      @relation(fields: [standardId], references: [id], onDelete: SetNull)
