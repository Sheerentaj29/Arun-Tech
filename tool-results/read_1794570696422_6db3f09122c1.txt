   182→// =====================================================
   183→// PIPE STANDARD REFERENCE DATA (ASME B36.10M / B36.19M)
   184→// =====================================================
   185→// Wall thickness values seeded from the publicly published ASME
   186→// reference tables for the most commonly used nominal pipe sizes and
   187→// schedules. These values MUST be re-verified against the licensed /
   188→// official ASME standard edition by an authorized administrator before
   189→// production use. An admin can edit, add, or delete entries, and every
   190→// edit is recorded with verifiedBy + verifiedAt + audit log entry.
   191→
   192→model PipeStandard {
   193→  id              String   @id @default(cuid())
   194→  standard        String   // "ASME B36.10M" | "ASME B36.19M"
   195→  edition         String   // e.g. "2018"
   196→  nominalSizeInch String   // e.g. "2" or "1/2"
   197→  nominalSizeMm   Float   // e.g. 60.3
   198→  dn              String   // e.g. "DN50"
   199→  schedule        String   // e.g. "SCH 40", "STD", "XS", "XXS", "SCH 40S"
   200→  wallThicknessMm Float   // nominal wall thickness in mm
   201→  oD_mm           Float?   // outside diameter (mm) when applicable
   202→  isVerified      Boolean  @default(false)
   203→  verifiedById    String?
   204→  verifiedAt      DateTime?
   205→  notes           String?
   206→  createdAt       DateTime @default(now())
   207→  updatedAt       DateTime @updatedAt
   208→
   209→  verifiedBy      User?    @relation("PipeStandardVerifiedBy", fields: [verifiedById], references: [id], onDelete: SetNull)
   210→  measurements    ThicknessMeasurement[]
   211→
   212→  @@unique([standard, edition, nominalSizeInch, schedule])
   213→  @@index([standard, nominalSizeInch])
   214→}

   576→// =====================================================
   577→// THICKNESS MEASUREMENT ROW
   578→// =====================================================
   579→// One row per measurement point. The first two columns (Dia mm (inch),
   580→// Pipe Schedule) are stored as structured values; the third column
   581→// (Nominal Thickness) is calculated from PipeStandard and may be
   582→// overridden only with a recorded reason. The four clock-position
   583→// measurements support an explicit status (Measured | NotMeasured |
   584→// NotApplicable) so missing data is never converted to zero.
   585→
   586→model ThicknessMeasurement {
   587→  id                    String   @id @default(cuid())
   588→  pipelineInspectionId String
   589→  pointNo               String   // free text, e.g. "P-1"
   590→  sortOrder             Int      @default(0)
   591→
   592→  // ─── Multi-table grouping (additive, default "default") ───
   593→  // When a Thickness report contains multiple tables, each row belongs to
   594→  // exactly one table group. Existing rows (and any new row in a
   595→  // single-table report) get "default" — so the existing single-table
   596→  // workflow keeps working unchanged. The list of table groups itself is
   597→  // stored as JSON on the parent PipelineInspection.thicknessTablesConfig.
   598→  tableGroup            String   @default("default")
   599→
   600→  // First two fixed columns
   601→  pipeSizeInch          String   // e.g. "2", "1/2", "6" (matches PipeStandard.nominalSizeInch)
   602→  pipeSchedule          String   // e.g. "SCH 40", "STD"
   603→
   604→  // Third column (calculated from standards; override captured separately)
   605→  nominalThicknessMm    Float?   // snapshot from PipeStandard at row creation/update
   606→  nominalOverride       Float?   // authorized override value (with reason at report level)
   607→  standardId            String?  // FK to PipeStandard used to compute nominalThicknessMm
   608→
   609→  // Fourth column
   610→  componentTypeId        String?
   611→  componentTypeCode      String   // snapshot code (PIPE | ELBOW | TEE | REDUCER | FLANGE | OTHER)
   612→
   613→  // Columns 5-9: clock-position measurements with explicit status
   614→  northMm               Float?
   615→  northStatus            String   @default("NotMeasured") // Measured | NotMeasured | NotApplicable
   616→  eastMm                 Float?
   617→  eastStatus             String   @default("NotMeasured")
   618→  southMm               Float?
   619→  southStatus            String   @default("NotMeasured")
   620→  westMm                 Float?
   621→  westStatus             String   @default("NotMeasured")
   622→
   623→  // Column 10: observed minimum (calculated; null if no valid measurements)
   624→  observedMinimumMm     Float?
   625→  observedMinimumOverride Float? // authorized override with reason
   626→
   627→  // Column 11: findings / comments
   628→  findingsComments      String?
   629→  // Custom column values stored as JSON: {"col_id_1": "5.2", "col_id_2": "N/A"}
   630→  // Each key corresponds to a custom column defined on the parent PipelineInspection.
   631→  customColumnValues    String?
   632→
   633→  createdAt             DateTime @default(now())
   634→  updatedAt             DateTime @updatedAt
   635→
   636→  pipelineInspection    PipelineInspection @relation(fields: [pipelineInspectionId], references: [id], onDelete: Cascade)
   637→  componentType         ComponentType?     @relation(fields: [componentTypeId], references: [id], onDelete: SetNull)
   638→  standard              PipeStandard?      @relation(fields: [standardId], references: [id], onDelete: SetNull)
   639→
   640→  @@index([pipelineInspectionId])
   641→  @@index([pipeSizeInch, pipeSchedule])
   642→}
