   152→// =====================================================
   153→// PIPE STANDARD REFERENCE DATA (ASME B36.10M / B36.19M)
   154→// =====================================================
   155→// Wall thickness values seeded from the publicly published ASME
   156→// reference tables for the most commonly used nominal pipe sizes and
   157→// schedules. These values MUST be re-verified against the licensed /
   158→// official ASME standard edition by an authorized administrator before
   159→// production use. An admin can edit, add, or delete entries, and every
   160→// edit is recorded with verifiedBy + verifiedAt + audit log entry.
   161→
   162→model PipeStandard {
   163→  id              String   @id @default(cuid())
   164→  standard        String   // "ASME B36.10M" | "ASME B36.19M"
   165→  edition         String   // e.g. "2018"
   166→  nominalSizeInch String   // e.g. "2" or "1/2"
   167→  nominalSizeMm   Float   // e.g. 60.3
   168→  dn              String   // e.g. "DN50"
   169→  schedule        String   // e.g. "SCH 40", "STD", "XS", "XXS", "SCH 40S"
   170→  wallThicknessMm Float   // nominal wall thickness in mm
   171→  oD_mm           Float?   // outside diameter (mm) when applicable
   172→  isVerified      Boolean  @default(false)
   173→  verifiedById    String?
   174→  verifiedAt      DateTime?
   175→  notes           String?
   176→  createdAt       DateTime @default(now())
   177→  updatedAt       DateTime @updatedAt
   178→
   179→  verifiedBy      User?    @relation("PipeStandardVerifiedBy", fields: [verifiedById], references: [id], onDelete: SetNull)
   180→  measurements    ThicknessMeasurement[]
   181→
   182→  @@unique([standard, edition, nominalSizeInch, schedule])
   183→  @@index([standard, nominalSizeInch])
   184→}

   449→// =====================================================
   450→// THICKNESS MEASUREMENT ROW
   451→// =====================================================
   452→// One row per measurement point. The first two columns (Dia mm (inch),
   453→// Pipe Schedule) are stored as structured values; the third column
   454→// (Nominal Thickness) is calculated from PipeStandard and may be
   455→// overridden only with a recorded reason. The four clock-position
   456→// measurements support an explicit status (Measured | NotMeasured |
   457→// NotApplicable) so missing data is never converted to zero.
   458→
   459→model ThicknessMeasurement {
   460→  id                    String   @id @default(cuid())
   461→  pipelineInspectionId String
   462→  pointNo               String   // free text, e.g. "P-1"
   463→  sortOrder             Int      @default(0)
   464→
   465→  // ─── Multi-table grouping (additive, default "default") ───
   466→  // When a Thickness report contains multiple tables, each row belongs to
   467→  // exactly one table group. Existing rows (and any new row in a
   468→  // single-table report) get "default" — so the existing single-table
   469→  // workflow keeps working unchanged. The list of table groups itself is
   470→  // stored as JSON on the parent PipelineInspection.thicknessTablesConfig.
   471→  tableGroup            String   @default("default")
   472→
   473→  // First two fixed columns
   474→  pipeSizeInch          String   // e.g. "2", "1/2", "6" (matches PipeStandard.nominalSizeInch)
   475→  pipeSchedule          String   // e.g. "SCH 40", "STD"
   476→
   477→  // Third column (calculated from standards; override captured separately)
   478→  nominalThicknessMm    Float?   // snapshot from PipeStandard at row creation/update
   479→  nominalOverride       Float?   // authorized override value (with reason at report level)
   480→  standardId            String?  // FK to PipeStandard used to compute nominalThicknessMm
   481→
   482→  // Fourth column
   483→  componentTypeId        String?
   484→  componentTypeCode      String   // snapshot code (PIPE | ELBOW | TEE | REDUCER | FLANGE | OTHER)
   485→
   486→  // Columns 5-9: clock-position measurements with explicit status
   487→  northMm               Float?
   488→  northStatus            String   @default("NotMeasured") // Measured | NotMeasured | NotApplicable
