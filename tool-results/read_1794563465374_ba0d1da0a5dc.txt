   182→// =====================================================
   183→// PIPE STANDARD REFERENCE DATA (ASME B36.10M / B36.19M)
   184→// =====================================================
   185→// Wall thickness values seeded from the publicly published ASME
   186→// reference tables for the most commonly used nominal pipe sizes and
   187→// schedules. These values MUST be re-verified against the licensed /
   188→// official ASME standard edition by an authorized administrator before
   189→// production use. An admin can edit, add, or delete entries, and every
   190→// edit is recorded with verifiedBy + verifiedAt + audit log entry.
   191→
   192→model PipeStandard {
   193→  id              String   @id @default(cuid())
   194→  standard        String   // "ASME B36.10M" | "ASME B36.19M"
   195→  edition         String   // e.g. "2018"
   196→  nominalSizeInch String   // e.g. "2" or "1/2"
   197→  nominalSizeMm   Float   // e.g. 60.3
   198→  dn              String   // e.g. "DN50"
   199→  schedule        String   // e.g. "SCH 40", "STD", "XS", "XXS", "SCH 40S"
   200→  wallThicknessMm Float   // nominal wall thickness in mm
   201→  oD_mm           Float?   // outside diameter (mm) when applicable
   202→  isVerified      Boolean  @default(false)
   203→  verifiedById    String?
   204→  verifiedAt      DateTime?
   205→  notes           String?
   206→  createdAt       DateTime @default(now())
   207→  updatedAt       DateTime @updatedAt
   208→
   209→  verifiedBy      User?    @relation("PipeStandardVerifiedBy", fields: [verifiedById], references: [id], onDelete: SetNull)
   210→  measurements    ThicknessMeasurement[]
   211→
   212→  @@unique([standard, edition, nominalSizeInch, schedule])
   213→  @@index([standard, nominalSizeInch])
   214→}

   505→// =====================================================
   506→// THICKNESS MEASUREMENT ROW
   507→// =====================================================
   508→// One row per measurement point. The first two columns (Dia mm (inch),
   509→// Pipe Schedule) are stored as structured values; the third column
   510→// (Nominal Thickness) is calculated from PipeStandard and may be
   511→// overridden only with a recorded reason. The four clock-position
   512→// measurements support an explicit status (Measured | NotMeasured |
   513→// NotApplicable) so missing data is never converted to zero.
   514→
   515→model ThicknessMeasurement {
   516→  id                    String   @id @default(cuid())
   517→  pipelineInspectionId String
   518→  pointNo               String   // free text, e.g. "P-1"
   519→  sortOrder             Int      @default(0)
   520→
   521→  // ─── Multi-table grouping (additive, default "default") ───
   522→  // When a Thickness report contains multiple tables, each row belongs to
   523→  // exactly one table group. Existing rows (and any new row in a
   524→  // single-table report) get "default" — so the existing single-table
   525→  // workflow keeps working unchanged. The list of table groups itself is
   526→  // stored as JSON on the parent PipelineInspection.thicknessTablesConfig.
   527→  tableGroup            String   @default("default")
   528→
   529→  // First two fixed columns
   530→  pipeSizeInch          String   // e.g. "2", "1/2", "6" (matches PipeStandard.nominalSizeInch)
   531→  pipeSchedule          String   // e.g. "SCH 40", "STD"
   532→
   533→  // Third column (calculated from standards; override captured separately)
   534→  nominalThicknessMm    Float?   // snapshot from PipeStandard at row creation/update
   535→  nominalOverride       Float?   // authorized override value (with reason at report level)
   536→  standardId            String?  // FK to PipeStandard used to compute nominalThicknessMm
   537→
   538→  // Fourth column
   539→  componentTypeId        String?
   540→  componentTypeCode      String   // snapshot code (PIPE | ELBOW | TEE | REDUCER | FLANGE | OTHER)
   541→
   542→  // Columns 5-9: clock-position measurements with explicit status
   543→  northMm               Float?
   544→  northStatus            String   @default("NotMeasured") // Measured | NotMeasured | NotApplicable
   545→  eastMm                 Float?
   546→  eastStatus             String   @default("NotMeasured")
   547→  southMm               Float?
   548→  southStatus            String   @default("NotMeasured")
   549→  westMm                 Float?
   550→  westStatus             String   @default("NotMeasured")
   551→
   552→  // Column 10: observed minimum (calculated; null if no valid measurements)
   553→  observedMinimumMm     Float?
   554→  observedMinimumOverride Float? // authorized override with reason
   555→
   556→  // Column 11: findings / comments
   557→  findingsComments      String?
   558→  // Custom column values stored as JSON: {"col_id_1": "5.2", "col_id_2": "N/A"}
   559→  // Each key corresponds to a custom column defined on the parent PipelineInspection.
   560→  customColumnValues    String?
   561→
   562→  createdAt             DateTime @default(now())
   563→  updatedAt             DateTime @updatedAt
   564→
   565→  pipelineInspection    PipelineInspection @relation(fields: [pipelineInspectionId], references: [id], onDelete: Cascade)
   566→  componentType         ComponentType?     @relation(fields: [componentTypeId], references: [id], onDelete: SetNull)
   567→  standard              PipeStandard?      @relation(fields: [standardId], references: [id], onDelete: SetNull)
   568→
   569→  @@index([pipelineInspectionId])
   570→  @@index([pipeSizeInch, pipeSchedule])
   571→}
