   182→// =====================================================
   183→// PIPE STANDARD REFERENCE DATA (ASME B36.10M / B36.19M)
   184→// =====================================================
   185→// Wall thickness values seeded from the publicly published ASME
   186→// reference tables for the most commonly used nominal pipe sizes and
   187→// schedules. These values MUST be re-verified against the licensed /
   188→// official ASME standard edition by an authorized administrator before
   189→// production use. An admin can edit, add, or delete entries, and every
   190→// edit is recorded with verifiedBy + verifiedAt + audit log entry.
   191→
   192→model PipeStandard {
   193→  id              String   @id @default(cuid())
   194→  standard        String   // "ASME B36.10M" | "ASME B36.19M"
   195→  edition         String   // e.g. "2018"
   196→  nominalSizeInch String   // e.g. "2" or "1/2"
   197→  nominalSizeMm   Float   // e.g. 60.3
   198→  dn              String   // e.g. "DN50"
   199→  schedule        String   // e.g. "SCH 40", "STD", "XS", "XXS", "SCH 40S"
   200→  wallThicknessMm Float   // nominal wall thickness in mm
   201→  oD_mm           Float?   // outside diameter (mm) when applicable
   202→  isVerified      Boolean  @default(false)
   203→  verifiedById    String?
   204→  verifiedAt      DateTime?
   205→  notes           String?
   206→  createdAt       DateTime @default(now())
   207→  updatedAt       DateTime @updatedAt
   208→
   209→  verifiedBy      User?    @relation("PipeStandardVerifiedBy", fields: [verifiedById], references: [id], onDelete: SetNull)
   210→  measurements    ThicknessMeasurement[]
   211→
   212→  @@unique([standard, edition, nominalSizeInch, schedule])
   213→  @@index([standard, nominalSizeInch])
   214→}

   380→// =====================================================
   381→// THICKNESS MEASUREMENT ROW
   382→// =====================================================
   383→// One row per measurement point. The first two columns (Dia mm (inch),
   384→// Pipe Schedule) are stored as structured values; the third column
   385→// (Nominal Thickness) is calculated from PipeStandard and may be
   386→// overridden only with a recorded reason. The four clock-position
   387→// measurements support an explicit status (Measured | NotMeasured |
   388→// NotApplicable) so missing data is never converted to zero.
   389→
   390→model ThicknessMeasurement {
   391→  id                    String   @id @default(cuid())
   392→  pipelineInspectionId String
   393→  pointNo               String   // free text, e.g. "P-1"
   394→  sortOrder             Int      @default(0)
   395→
   396→  // First two fixed columns
   397→  pipeSizeInch          String   // e.g. "2", "1/2", "6" (matches PipeStandard.nominalSizeInch)
   398→  pipeSchedule          String   // e.g. "SCH 40", "STD"
   399→
   400→  // Third column (calculated from standards; override captured separately)
   401→  nominalThicknessMm    Float?   // snapshot from PipeStandard at row creation/update
   402→  nominalOverride       Float?   // authorized override value (with reason at report level)
   403→  standardId            String?  // FK to PipeStandard used to compute nominalThicknessMm
   404→
   405→  // Fourth column
   406→  componentTypeId        String?
   407→  componentTypeCode      String   // snapshot code (PIPE | ELBOW | TEE | REDUCER | FLANGE | OTHER)
   408→
   409→  // Columns 5-9: clock-position measurements with explicit status
   410→  northMm               Float?
   411→  northStatus            String   @default("NotMeasured") // Measured | NotMeasured | NotApplicable
   412→  eastMm                 Float?
   413→  eastStatus             String   @default("NotMeasured")
   414→  southMm               Float?
   415→  southStatus            String   @default("NotMeasured")
   416→  westMm                 Float?
   417→  westStatus             String   @default("NotMeasured")
   418→
   419→  // Column 10: observed minimum (calculated; null if no valid measurements)
   420→  observedMinimumMm     Float?
   421→  observedMinimumOverride Float? // authorized override with reason
   422→
   423→  // Column 11: findings / comments
   424→  findingsComments      String?
   425→  // Custom column values stored as JSON: {"col_id_1": "5.2", "col_id_2": "N/A"}
   426→  // Each key corresponds to a custom column defined on the parent PipelineInspection.
   427→  customColumnValues    String?
   428→
   429→  createdAt             DateTime @default(now())
   430→  updatedAt             DateTime @updatedAt
   431→
   432→  pipelineInspection    PipelineInspection @relation(fields: [pipelineInspectionId], references: [id], onDelete: Cascade)
   433→  componentType         ComponentType?     @relation(fields: [componentTypeId], references: [id], onDelete: SetNull)
   434→  standard              PipeStandard?      @relation(fields: [standardId], references: [id], onDelete: SetNull)
   435→
   436→  @@index([pipelineInspectionId])
   437→  @@index([pipeSizeInch, pipeSchedule])
   438→}

   502→  approvedByName   String?
   503→  createdAt         DateTime @default(now())

   528→  caption              String?  // optional: "Overall pipeline view", "Corrosion observed", etc.
