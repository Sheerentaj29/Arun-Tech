   182→// =====================================================
   183→// PIPE STANDARD REFERENCE DATA (ASME B36.10M / B36.19M)
   184→// =====================================================
   185→// Wall thickness values seeded from the publicly published ASME
   186→// reference tables for the most commonly used nominal pipe sizes and
   187→// schedules. These values MUST be re-verified against the licensed /
   188→// official ASME standard edition by an authorized administrator before
   189→// production use. An admin can edit, add, or delete entries, and every
   190→// edit is recorded with verifiedBy + verifiedAt + audit log entry.
   191→
   192→model PipeStandard {
   193→  id              String   @id @default(cuid())
   194→  standard        String   // "ASME B36.10M" | "ASME B36.19M"
   195→  edition         String   // e.g. "2018"
   196→  nominalSizeInch String   // e.g. "2" or "1/2"
   197→  nominalSizeMm   Float   // e.g. 60.3
   198→  dn              String   // e.g. "DN50"
   199→  schedule        String   // e.g. "SCH 40", "STD", "XS", "XXS", "SCH 40S"
   200→  wallThicknessMm Float   // nominal wall thickness in mm
   201→  oD_mm           Float?   // outside diameter (mm) when applicable
   202→  isVerified      Boolean  @default(false)
   203→  verifiedById    String?
   204→  verifiedAt      DateTime?
   205→  notes           String?
   206→  createdAt       DateTime @default(now())
   207→  updatedAt       DateTime @updatedAt
   208→
   209→  verifiedBy      User?    @relation("PipeStandardVerifiedBy", fields: [verifiedById], references: [id], onDelete: SetNull)
   210→  measurements    ThicknessMeasurement[]
   211→
   212→  @@unique([standard, edition, nominalSizeInch, schedule])
   213→  @@index([standard, nominalSizeInch])
   214→}

   529→// =====================================================
   530→// THICKNESS MEASUREMENT ROW
   531→// =====================================================
   532→// One row per measurement point. The first two columns (Dia mm (inch),
   533→// Pipe Schedule) are stored as structured values; the third column
   534→// (Nominal Thickness) is calculated from PipeStandard and may be
   535→// overridden only with a recorded reason. The four clock-position
   536→// measurements support an explicit status (Measured | NotMeasured |
   537→// NotApplicable) so missing data is never converted to zero.
   538→
   539→model ThicknessMeasurement {
   540→  id                    String   @id @default(cuid())
   541→  pipelineInspectionId String
   542→  pointNo               String   // free text, e.g. "P-1"
   543→  sortOrder             Int      @default(0)
   544→
   545→  // ─── Multi-table grouping (additive, default "default") ───
   546→  // When a Thickness report contains multiple tables, each row belongs to
   547→  // exactly one table group. Existing rows (and any new row in a
   548→  // single-table report) get "default" — so the existing single-table
   549→  // workflow keeps working unchanged. The list of table groups itself is
   550→  // stored as JSON on the parent PipelineInspection.thicknessTablesConfig.
   551→  tableGroup            String   @default("default")
   552→
   553→  // First two fixed columns
   554→  pipeSizeInch          String   // e.g. "2", "1/2", "6" (matches PipeStandard.nominalSizeInch)
   555→  pipeSchedule          String   // e.g. "SCH 40", "STD"
   556→
   557→  // Third column (calculated from standards; override captured separately)
   558→  nominalThicknessMm    Float?   // snapshot from PipeStandard at row creation/update
   559→  nominalOverride       Float?   // authorized override value (with reason at report level)
   560→  standardId            String?  // FK to PipeStandard used to compute nominalThicknessMm
   561→
   562→  // Fourth column
   563→  componentTypeId        String?
   564→  componentTypeCode      String   // snapshot code (PIPE | ELBOW | TEE | REDUCER | FLANGE | OTHER)
   565→
   566→  // Columns 5-9: clock-position measurements with explicit status
   567→  northMm               Float?
   568→  northStatus            String   @default("NotMeasured") // Measured | NotMeasured | NotApplicable
   569→  eastMm                 Float?
   570→  eastStatus             String   @default("NotMeasured")
   571→  southMm               Float?
   572→  southStatus            String   @default("NotMeasured")
   573→  westMm                 Float?
   574→  westStatus             String   @default("NotMeasured")
   575→
   576→  // Column 10: observed minimum (calculated; null if no valid measurements)
   577→  observedMinimumMm     Float?
   578→  observedMinimumOverride Float? // authorized override with reason
   579→
   580→  // Column 11: findings / comments
   581→  findingsComments      String?
   582→  // Custom column values stored as JSON: {"col_id_1": "5.2", "col_id_2": "N/A"}
   583→  // Each key corresponds to a custom column defined on the parent PipelineInspection.
   584→  customColumnValues    String?
   585→
   586→  createdAt             DateTime @default(now())
   587→  updatedAt             DateTime @updatedAt
   588→
   589→  pipelineInspection    PipelineInspection @relation(fields: [pipelineInspectionId], references: [id], onDelete: Cascade)
   590→  componentType         ComponentType?     @relation(fields: [componentTypeId], references: [id], onDelete: SetNull)
   591→  standard              PipeStandard?      @relation(fields: [standardId], references: [id], onDelete: SetNull)
   592→
   593→  @@index([pipelineInspectionId])
   594→  @@index([pipeSizeInch, pipeSchedule])
   595→}
