   328→   328→export function PipelineReportDetailView() {
   329→   329→  const {
   330→   330→    user,
   331→   331→    selectedPipelineReportId,
   332→   332→    setView,
   333→   333→    openEditPipelineReport,
   334→   334→    refresh,
   335→   335→    refreshKey,
   336→   336→  } = useAppStore();
   337→   337→  const [report, setReport] = useState<FullReport | null>(null);
   338→   338→  const [loading, setLoading] = useState(true);
   339→   339→  const [error, setError] = useState<string | null>(null);
   340→   340→  const [acting, setActing] = useState(false);
   341→   341→  // Dialog state for workflow actions (submit / approve / return / revise)
   342→   342→  const [dialogMode, setDialogMode] = useState<null | "submit" | "approve" | "return" | "revise">(null);
   343→   343→  const [dialogText, setDialogText] = useState("");
   344→   344→  // Lightbox state — when set, the image at this URL is shown full-screen
   345→   345→  const [lightbox, setLightbox] = useState<{ url: string; caption: string } | null>(null);
   346→   346→
   347→   347→  const load = useCallback(() => {
   348→   348→    if (!selectedPipelineReportId) return;
   349→   349→    setLoading(true);
   350→   350→    apiFetch<FullReport>(`/api/pipeline-inspections/${selectedPipelineReportId}`)
   351→   351→      .then((r) => setReport(r))
   352→   352→      .catch((e) => setError(e.message))
   353→   353→      .finally(() => setLoading(false));
   354→   354→  }, [selectedPipelineReportId]);
   355→   355→
   356→   356→  useEffect(() => { load(); }, [load, refreshKey]);
   357→   357→
   358→   358→  // Role-based permissions (derived from session user + current report state)
   359→   359→  const isAdmin = user?.dashboardRole === "ADMIN";
   360→   360→  const isOwner = !!(user && report && report.inspectorId === user.id);
   361→   361→  const status = report?.status;
   362→   362→  // Employee can edit header/measurements only on their own Draft / Returned reports
   363→   363→  const canEditDraft =
   364→   364→    !!user && !!report && (
   365→   365→      isAdmin ||
   366→   366→      (isOwner && (status === "Draft" || status === "Returned for Revision"))
   367→   367→    );
   368→   368→  // Employee can submit (only their own Draft / Returned reports)
   369→   369→  const canSubmit =
   370→   370→    !!user && !!report && !isAdmin && isOwner &&
   371→   371→    (status === "Draft" || status === "Returned for Revision");
   372→   372→  // Employee can revise their own Approved report (creates new revision)
   373→   373→  const canRevise =
   374→   374→    !!user && !!report && !isAdmin && isOwner && status === "Approved";
   375→   375→  // Admin can approve or return when status is Submitted for Review or Changes Pending Review
   376→   376→  const canApproveOrReturn =
   377→   377→    !!user && !!report && isAdmin &&
   378→   378→    (status === "Submitted for Review" || status === "Changes Pending Review");
   379→   379→
   380→   380→  function openDialog(mode: "submit" | "approve" | "return" | "revise") {
   381→   381→    setDialogMode(mode);
   382→   382→    setDialogText("");
   383→   383→  }
   384→   384→
   385→   385→  async function executeWorkflow() {
   386→   386→    if (!report || !dialogMode) return;
   387→   387→    setActing(true);
   388→   388→    try {
   389→   389→      const body = dialogText.trim()
   390→   390→        ? (dialogMode === "submit"
   391→   391→            ? { submissionNotes: dialogText }
   392→   392→            : dialogMode === "approve"
   393→   393→              ? { approvalNotes: dialogText }
   394→   394→              : dialogMode === "return"
   395→   395→                ? { reviewComments: dialogText }
   396→   396→                : { changeDescription: dialogText })
   397→   397→        : {};
   398→   398→      const endpoint =
   399→   399→        dialogMode === "submit" ? "submit" :
   400→   400→        dialogMode === "approve" ? "approve" :
   401→   401→        dialogMode === "return" ? "return" :
   402→   402→        "revise";
   403→   403→      await apiFetch(`/api/pipeline-inspections/${report.id}/${endpoint}`, {
   404→   404→        method: "POST",
   405→   405→        body: JSON.stringify(body),
   406→   406→      });
   407→   407→      const labels: Record<string, string> = {
   408→   408→        submit: "Submitted for review",
   409→   409→        approve: "Approved",
   410→   410→        return: "Returned for revision",
   411→   411→        revise: "Revision started",
   412→   412→      };
   413→   413→      toast.success(labels[dialogMode]);
   414→   414→      setDialogMode(null);
   415→   415→      setDialogText("");
   416→   416→      refresh();
   417→   417→      load();
   418→   418→    } catch (e) {
   419→   419→      const err = e as { message?: string };
   420→   420→      toast.error(err.message ?? "Action failed");
   421→   421→    } finally {
   422→   422→      setActing(false);
   423→   423→    }
   424→   424→  }
   425→   425→
   426→   426→  async function downloadPdf(orientation?: "landscape" | "portrait") {
   427→   427→    if (!report) return;
   428→   428→    const orient = orientation ?? "landscape";
   429→   429→    try {
   430→   430→      // Use fetch() instead of window.open() so the Authorization header
   431→   431→      // (from localStorage) is sent along with the cookie. This fixes the
   432→   432→      // download failure caused by popup blockers and missing auth headers.
   433→   433→      const token = typeof window !== "undefined" ? window.localStorage.getItem("ira_session_token") : null;
   434→   434→      const res = await fetch(`/api/pipeline-inspections/${report.id}/pdf?orientation=${orient}`, {
   435→   435→        credentials: "include",
   436→   436→        headers: token ? { Authorization: `Bearer ${token}` } : {},
   437→   437→      });
   438→   438→      if (!res.ok) {
   439→   439→        const body = await res.json().catch(() => ({}));
   440→   440→        throw new Error(body?.error ?? `Download failed (HTTP ${res.status})`);
   441→   441→      }
   442→   442→      const blob = await res.blob();
   443→   443→      const url = URL.createObjectURL(blob);
   444→   444→      const a = document.createElement("a");
   445→   445→      a.href = url;
   446→   446→      // Build a descriptive filename
   447→   447→      const filename = `Inspection_Report_${report.reportNumber.replace(/[^a-zA-Z0-9-]/g, "_")}_${orient}.pdf`;
   448→   448→      a.download = filename;
   449→   449→      document.body.appendChild(a);
   450→   450→      a.click();
   451→   451→      a.remove();
   452→   452→      URL.revokeObjectURL(url);
   453→   453→      toast.success(`Downloaded ${filename}`);
   454→   454→    } catch (e) {
   455→   455→      const err = e as Error;
   456→   456→      toast.error(`PDF download failed: ${err.message}`);
   457→   457→    }
   458→   458→  }
   459→   459→
   460→   460→  if (loading) {
   461→   461→    return (
   462→   462→      <div className="p-8 text-muted-foreground flex items-center gap-2">
   463→   463→        <Loader2 className="h-4 w-4 animate-spin" /> Loading report...
