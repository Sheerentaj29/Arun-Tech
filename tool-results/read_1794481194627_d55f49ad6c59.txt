   328→export function PipelineReportDetailView() {
   329→  const {
   330→    user,
   331→    selectedPipelineReportId,
   332→    setView,
   333→    openEditPipelineReport,
   334→    refresh,
   335→    refreshKey,
   336→  } = useAppStore();
   337→  const [report, setReport] = useState<FullReport | null>(null);
   338→  const [loading, setLoading] = useState(true);
   339→  const [error, setError] = useState<string | null>(null);
   340→  const [acting, setActing] = useState(false);
   341→  // Dialog state for workflow actions (submit / approve / return / revise)
   342→  const [dialogMode, setDialogMode] = useState<null | "submit" | "approve" | "return" | "revise">(null);
   343→  const [dialogText, setDialogText] = useState("");
   344→  // Lightbox state — when set, the image at this URL is shown full-screen
   345→  const [lightbox, setLightbox] = useState<{ url: string; caption: string } | null>(null);
   346→
   347→  const load = useCallback(() => {
   348→    if (!selectedPipelineReportId) return;
   349→    setLoading(true);
   350→    apiFetch<FullReport>(`/api/pipeline-inspections/${selectedPipelineReportId}`)
   351→      .then((r) => setReport(r))
   352→      .catch((e) => setError(e.message))
   353→      .finally(() => setLoading(false));
   354→  }, [selectedPipelineReportId]);
   355→
   356→  useEffect(() => { load(); }, [load, refreshKey]);
   357→
   358→  // Role-based permissions (derived from session user + current report state)
   359→  const isAdmin = user?.dashboardRole === "ADMIN";
   360→  const isOwner = !!(user && report && report.inspectorId === user.id);
   361→  const status = report?.status;
   362→  // Employee can edit header/measurements only on their own Draft / Returned reports
   363→  const canEditDraft =
   364→    !!user && !!report && (
   365→      isAdmin ||
   366→      (isOwner && (status === "Draft" || status === "Returned for Revision"))
   367→    );
   368→  // Employee can submit (only their own Draft / Returned reports)
   369→  const canSubmit =
   370→    !!user && !!report && !isAdmin && isOwner &&
   371→    (status === "Draft" || status === "Returned for Revision");
   372→  // Employee can revise their own Approved report (creates new revision)
   373→  const canRevise =
   374→    !!user && !!report && !isAdmin && isOwner && status === "Approved";
   375→  // Admin can approve or return when status is Submitted for Review or Changes Pending Review
   376→  const canApproveOrReturn =
   377→    !!user && !!report && isAdmin &&
   378→    (status === "Submitted for Review" || status === "Changes Pending Review");
   379→
   380→  function openDialog(mode: "submit" | "approve" | "return" | "revise") {
   381→    setDialogMode(mode);
   382→    setDialogText("");
   383→  }
   384→
   385→  async function executeWorkflow() {
   386→    if (!report || !dialogMode) return;
   387→    setActing(true);
   388→    try {
   389→      const body = dialogText.trim()
   390→        ? (dialogMode === "submit"
   391→            ? { submissionNotes: dialogText }
   392→            : dialogMode === "approve"
   393→              ? { approvalNotes: dialogText }
   394→              : dialogMode === "return"
   395→                ? { reviewComments: dialogText }
   396→                : { changeDescription: dialogText })
   397→        : {};
   398→      const endpoint =
   399→        dialogMode === "submit" ? "submit" :
   400→        dialogMode === "approve" ? "approve" :
   401→        dialogMode === "return" ? "return" :
   402→        "revise";
   403→      await apiFetch(`/api/pipeline-inspections/${report.id}/${endpoint}`, {
   404→        method: "POST",
   405→        body: JSON.stringify(body),
   406→      });
   407→      const labels: Record<string, string> = {
   408→        submit: "Submitted for review",
   409→        approve: "Approved",
   410→        return: "Returned for revision",
   411→        revise: "Revision started",
   412→      };
   413→      toast.success(labels[dialogMode]);
   414→      setDialogMode(null);
   415→      setDialogText("");
   416→      refresh();
   417→      load();
   418→    } catch (e) {
   419→      const err = e as { message?: string };
   420→      toast.error(err.message ?? "Action failed");
   421→    } finally {
   422→      setActing(false);
   423→    }
   424→  }
   425→
   426→  async function downloadPdf(orientation?: "landscape" | "portrait") {
   427→    if (!report) return;
   428→    const orient = orientation ?? "landscape";
   429→    try {
   430→      // Use fetch() instead of window.open() so the Authorization header
   431→      // (from localStorage) is sent along with the cookie. This fixes the
   432→      // download failure caused by popup blockers and missing auth headers.
   433→      const token = typeof window !== "undefined" ? window.localStorage.getItem("ira_session_token") : null;
   434→      const res = await fetch(`/api/pipeline-inspections/${report.id}/pdf?orientation=${orient}`, {
   435→        credentials: "include",
   436→        headers: token ? { Authorization: `Bearer ${token}` } : {},
   437→      });
   438→      if (!res.ok) {
   439→        const body = await res.json().catch(() => ({}));
   440→        throw new Error(body?.error ?? `Download failed (HTTP ${res.status})`);
   441→      }
   442→      const blob = await res.blob();
   443→      const url = URL.createObjectURL(blob);
   444→      const a = document.createElement("a");
   445→      a.href = url;
   446→      // Build a descriptive filename
   447→      const filename = `Inspection_Report_${report.reportNumber.replace(/[^a-zA-Z0-9-]/g, "_")}_${orient}.pdf`;
   448→      a.download = filename;
   449→      document.body.appendChild(a);
   450→      a.click();
   451→      a.remove();
   452→      URL.revokeObjectURL(url);
   453→      toast.success(`Downloaded ${filename}`);
   454→    } catch (e) {
   455→      const err = e as Error;
   456→      toast.error(`PDF download failed: ${err.message}`);
   457→    }
   458→  }
   459→
   460→  if (loading) {
   461→    return (
   462→      <div className="p-8 text-muted-foreground flex items-center gap-2">
   463→        <Loader2 className="h-4 w-4 animate-spin" /> Loading report...
   464→      </div>
   465→    );
   466→  }
   467→  if (error) {
   468→    return (
   469→      <Alert variant="destructive">
   470→        <AlertTitle>Failed to load</AlertTitle>
   471→        <AlertDescription>{error}</AlertDescription>
   472→      </Alert>
   473→    );
   474→  }
   475→  if (!report) return <p className="p-8 text-muted-foreground">No report selected.</p>;
   476→
   477→  const allVerified = report.measurements.every(
   478→    (m) => m.standard?.isVerified || m.standard === null,
   479→  );
   480→
   481→  const dialogTitle: Record<string, string> = {
   482→    submit: "Submit Report for Review",
   483→    approve: "Approve Report",
   484→    return: "Return Report for Revision",
   485→    revise: "Start New Revision",
   486→  };
   487→  const dialogDesc: Record<string, string> = {
   488→    submit: "Submit this report to the administrator for approval. You won't be able to edit it until it's approved or returned.",
   489→    approve: "Approve this report. The current state will be saved as an approved revision; previous approved revisions will be marked as Superseded but remain available.",
   490→    return: "Return this report to the employee for correction. Mandatory review comments (at least 10 chars) will be visible to them.",
   491→    revise: "Start a new revision of this approved report. The previously-approved version stays available; this new revision will need admin approval before becoming the current approved version.",
   492→  };
   493→  const dialogLabel: Record<string, string> = {
   494→    submit: "Submission notes (optional)",
   495→    approve: "Approval notes (optional)",
   496→    return: "Review comments (REQUIRED, >= 10 chars)",
   497→    revise: "What are you changing? (REQUIRED, >= 5 chars)",
   498→  };
   499→  const dialogActionLabel: Record<string, string> = {
   500→    submit: "Submit for Review",
   501→    approve: "Approve",
   502→    return: "Return for Revision",
   503→    revise: "Start Revision",
   504→  };
   505→
   506→  return (
   507→    <div className="space-y-4">
   508→      {/* Top bar */}
   509→      <div className="flex items-center justify-between">
   510→        <Button variant="ghost" size="sm" onClick={() => setView("pipeline-reports")}>
   511→          <ArrowLeft className="mr-2 h-4 w-4" /> Back to list
   512→        </Button>
   513→        <div className="flex gap-2">
   514→          <Button variant="outline" size="sm" onClick={() => downloadPdf("landscape")}>
   515→            <Download className="mr-2 h-4 w-4" /> PDF (Landscape)
   516→          </Button>
   517→          <Button variant="outline" size="sm" onClick={() => downloadPdf("portrait")}>
   518→            <Download className="mr-2 h-4 w-4" /> PDF (Portrait)
   519→          </Button>
   520→          {canEditDraft && (
   521→            <Button size="sm" onClick={() => openEditPipelineReport(report.id)}>
   522→              <Pencil className="mr-2 h-4 w-4" /> Edit
   523→            </Button>
   524→          )}
   525→          {canSubmit && (
   526→            <Button size="sm" onClick={() => openDialog("submit")}>
   527→              <Send className="mr-2 h-4 w-4" /> Submit for Review
   528→            </Button>
   529→          )}
   530→          {canRevise && (
   531→            <Button size="sm" variant="default" onClick={() => openDialog("revise")}>
   532→              <GitBranch className="mr-2 h-4 w-4" /> Start New Revision
   533→            </Button>
   534→          )}
   535→          {canApproveOrReturn && (
   536→            <>
   537→              <Button size="sm" variant="default" onClick={() => openDialog("approve")}>
   538→                <CheckCircle2 className="mr-2 h-4 w-4" /> Approve
   539→              </Button>
   540→              <Button size="sm" variant="outline" className="text-red-700 border-red-300" onClick={() => openDialog("return")}>
   541→                <RotateCcw className="mr-2 h-4 w-4" /> Return for Revision
   542→              </Button>
   543→            </>
   544→          )}
   545→        </div>
   546→      </div>
   547→
   548→      {/* Status banner */}
   549→      <div className="flex flex-wrap items-center gap-3">
