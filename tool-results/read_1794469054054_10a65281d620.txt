   334→export function PipelineReportDetailView() {
   335→  const {
   336→    user,
   337→    selectedPipelineReportId,
   338→    setView,
   339→    openEditPipelineReport,
   340→    refresh,
   341→    refreshKey,
   342→  } = useAppStore();
   343→  const [report, setReport] = useState<FullReport | null>(null);
   344→  const [loading, setLoading] = useState(true);
   345→  const [error, setError] = useState<string | null>(null);
   346→  const [acting, setActing] = useState(false);
   347→  // Dialog state for workflow actions (submit / approve / return / revise)
   348→  const [dialogMode, setDialogMode] = useState<null | "submit" | "approve" | "return" | "revise">(null);
   349→  const [dialogText, setDialogText] = useState("");
   350→  // Lightbox state — when set, the image at this URL is shown full-screen
   351→  const [lightbox, setLightbox] = useState<{ url: string; caption: string } | null>(null);
   352→
   353→  const load = useCallback(() => {
   354→    if (!selectedPipelineReportId) return;
   355→    setLoading(true);
   356→    apiFetch<FullReport>(`/api/pipeline-inspections/${selectedPipelineReportId}`)
   357→      .then((r) => setReport(r))
   358→      .catch((e) => setError(e.message))
   359→      .finally(() => setLoading(false));
   360→  }, [selectedPipelineReportId]);
   361→
   362→  useEffect(() => { load(); }, [load, refreshKey]);
   363→
   364→  // Role-based permissions (derived from session user + current report state)
   365→  const isAdmin = user?.dashboardRole === "ADMIN";
   366→  const isOwner = !!(user && report && report.inspectorId === user.id);
   367→  const status = report?.status;
   368→  // Employee can edit header/measurements only on their own Draft / Returned reports
   369→  const canEditDraft =
   370→    !!user && !!report && (
   371→      isAdmin ||
   372→      (isOwner && (status === "Draft" || status === "Returned for Revision"))
   373→    );
   374→  // Employee can submit (only their own Draft / Returned reports)
   375→  const canSubmit =
   376→    !!user && !!report && !isAdmin && isOwner &&
   377→    (status === "Draft" || status === "Returned for Revision");
   378→  // Employee can revise their own Approved report (creates new revision)
   379→  const canRevise =
   380→    !!user && !!report && !isAdmin && isOwner && status === "Approved";
   381→  // Admin can approve or return when status is Submitted for Review or Changes Pending Review
   382→  const canApproveOrReturn =
   383→    !!user && !!report && isAdmin &&
   384→    (status === "Submitted for Review" || status === "Changes Pending Review");
   385→
   386→  function openDialog(mode: "submit" | "approve" | "return" | "revise") {
   387→    setDialogMode(mode);
   388→    setDialogText("");
   389→  }
   390→
   391→  async function executeWorkflow() {
   392→    if (!report || !dialogMode) return;
   393→    setActing(true);
   394→    try {
   395→      const body = dialogText.trim()
   396→        ? (dialogMode === "submit"
   397→            ? { submissionNotes: dialogText }
   398→            : dialogMode === "approve"
   399→              ? { approvalNotes: dialogText }
   400→              : dialogMode === "return"
   401→                ? { reviewComments: dialogText }
   402→                : { changeDescription: dialogText })
   403→        : {};
   404→      const endpoint =
   405→        dialogMode === "submit" ? "submit" :
   406→        dialogMode === "approve" ? "approve" :
   407→        dialogMode === "return" ? "return" :
   408→        "revise";
   409→      await apiFetch(`/api/pipeline-inspections/${report.id}/${endpoint}`, {
   410→        method: "POST",
   411→        body: JSON.stringify(body),
   412→      });
   413→      const labels: Record<string, string> = {
   414→        submit: "Submitted for review",
   415→        approve: "Approved",
   416→        return: "Returned for revision",
   417→        revise: "Revision started",
   418→      };
   419→      toast.success(labels[dialogMode]);
   420→      setDialogMode(null);
   421→      setDialogText("");
   422→      refresh();
   423→      load();
   424→    } catch (e) {
   425→      const err = e as { message?: string };
   426→      toast.error(err.message ?? "Action failed");
   427→    } finally {
   428→      setActing(false);
   429→    }
   430→  }
   431→
   432→  async function downloadPdf(orientation?: "landscape" | "portrait") {
   433→    if (!report) return;
   434→    const orient = orientation ?? "landscape";
   435→    try {
   436→      // Use fetch() instead of window.open() so the Authorization header
   437→      // (from localStorage) is sent along with the cookie. This fixes the
   438→      // download failure caused by popup blockers and missing auth headers.
   439→      const token = typeof window !== "undefined" ? window.localStorage.getItem("ira_session_token") : null;
   440→      const res = await fetch(`/api/pipeline-inspections/${report.id}/pdf?orientation=${orient}`, {
   441→        credentials: "include",
   442→        headers: token ? { Authorization: `Bearer ${token}` } : {},
   443→      });
   444→      if (!res.ok) {
   445→        const body = await res.json().catch(() => ({}));
   446→        throw new Error(body?.error ?? `Download failed (HTTP ${res.status})`);
   447→      }
   448→      const blob = await res.blob();
   449→      const url = URL.createObjectURL(blob);
   450→      const a = document.createElement("a");
   451→      a.href = url;
   452→      // Build a descriptive filename
   453→      const filename = `Inspection_Report_${report.reportNumber.replace(/[^a-zA-Z0-9-]/g, "_")}_${orient}.pdf`;
   454→      a.download = filename;
   455→      document.body.appendChild(a);
   456→      a.click();
   457→      a.remove();
   458→      URL.revokeObjectURL(url);
   459→      toast.success(`Downloaded ${filename}`);
   460→    } catch (e) {
   461→      const err = e as Error;
   462→      toast.error(`PDF download failed: ${err.message}`);
   463→    }
   464→  }
   465→
   466→  if (loading) {
   467→    return (
   468→      <div className="p-8 text-muted-foreground flex items-center gap-2">
   469→        <Loader2 className="h-4 w-4 animate-spin" /> Loading report...
   470→      </div>
   471→    );
   472→  }
   473→  if (error) {
   474→    return (
   475→      <Alert variant="destructive">
   476→        <AlertTitle>Failed to load</AlertTitle>
   477→        <AlertDescription>{error}</AlertDescription>
   478→      </Alert>
   479→    );
   480→  }
   481→  if (!report) return <p className="p-8 text-muted-foreground">No report selected.</p>;
   482→
   483→  const allVerified = report.measurements.every(
   484→    (m) => m.standard?.isVerified || m.standard === null,
   485→  );
   486→
   487→  const dialogTitle: Record<string, string> = {
   488→    submit: "Submit Report for Review",
   489→    approve: "Approve Report",
   490→    return: "Return Report for Revision",
   491→    revise: "Start New Revision",
   492→  };
   493→  const dialogDesc: Record<string, string> = {
   494→    submit: "Submit this report to the administrator for approval. You won't be able to edit it until it's approved or returned.",
   495→    approve: "Approve this report. The current state will be saved as an approved revision; previous approved revisions will be marked as Superseded but remain available.",
   496→    return: "Return this report to the employee for correction. Mandatory review comments (at least 10 chars) will be visible to them.",
   497→    revise: "Start a new revision of this approved report. The previously-approved version stays available; this new revision will need admin approval before becoming the current approved version.",
   498→  };
   499→  const dialogLabel: Record<string, string> = {
   500→    submit: "Submission notes (optional)",
   501→    approve: "Approval notes (optional)",
   502→    return: "Review comments (REQUIRED, >= 10 chars)",
   503→    revise: "What are you changing? (REQUIRED, >= 5 chars)",
   504→  };
   505→  const dialogActionLabel: Record<string, string> = {
   506→    submit: "Submit for Review",
   507→    approve: "Approve",
   508→    return: "Return for Revision",
   509→    revise: "Start Revision",
   510→  };
   511→
   512→  return (
   513→    <div className="space-y-4">
   514→      {/* Top bar */}
   515→      <div className="flex items-center justify-between">
   516→        <Button variant="ghost" size="sm" onClick={() => setView("pipeline-reports")}>
   517→          <ArrowLeft className="mr-2 h-4 w-4" /> Back to list
   518→        </Button>
   519→        <div className="flex gap-2">
   520→          <Button variant="outline" size="sm" onClick={() => downloadPdf("landscape")}>
   521→            <Download className="mr-2 h-4 w-4" /> PDF (Landscape)
   522→          </Button>
   523→          <Button variant="outline" size="sm" onClick={() => downloadPdf("portrait")}>
   524→            <Download className="mr-2 h-4 w-4" /> PDF (Portrait)
   525→          </Button>
   526→          {canEditDraft && (
   527→            <Button size="sm" onClick={() => openEditPipelineReport(report.id)}>
   528→              <Pencil className="mr-2 h-4 w-4" /> Edit
   529→            </Button>
   530→          )}
   531→          {canSubmit && (
   532→            <Button size="sm" onClick={() => openDialog("submit")}>
   533→              <Send className="mr-2 h-4 w-4" /> Submit for Review
   534→            </Button>
   535→          )}
   536→          {canRevise && (
   537→            <Button size="sm" variant="default" onClick={() => openDialog("revise")}>
   538→              <GitBranch className="mr-2 h-4 w-4" /> Start New Revision
   539→            </Button>
   540→          )}
   541→          {canApproveOrReturn && (
   542→            <>
   543→              <Button size="sm" variant="default" onClick={() => openDialog("approve")}>
   544→                <CheckCircle2 className="mr-2 h-4 w-4" /> Approve
   545→              </Button>
   546→              <Button size="sm" variant="outline" className="text-red-700 border-red-300" onClick={() => openDialog("return")}>
   547→                <RotateCcw className="mr-2 h-4 w-4" /> Return for Revision
   548→              </Button>
   549→            </>
