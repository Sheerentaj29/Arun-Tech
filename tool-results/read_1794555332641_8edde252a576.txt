    95→// =====================================================
    96→// PIPE STANDARD REFERENCE DATA (ASME B36.10M / B36.19M)
    97→// =====================================================
    98→// Wall thickness values seeded from the publicly published ASME
    99→// reference tables for the most commonly used nominal pipe sizes and
   100→// schedules. These values MUST be re-verified against the licensed /
   101→// official ASME standard edition by an authorized administrator before
   102→// production use. An admin can edit, add, or delete entries, and every
   103→// edit is recorded with verifiedBy + verifiedAt + audit log entry.
   104→
   105→model PipeStandard {
   106→  id              String   @id @default(cuid())
   107→  standard        String   // "ASME B36.10M" | "ASME B36.19M"
   108→  edition         String   // e.g. "2018"
   109→  nominalSizeInch String   // e.g. "2" or "1/2"
   110→  nominalSizeMm   Float   // e.g. 60.3
   111→  dn              String   // e.g. "DN50"
   112→  schedule        String   // e.g. "SCH 40", "STD", "XS", "XXS", "SCH 40S"
   113→  wallThicknessMm Float   // nominal wall thickness in mm
   114→  oD_mm           Float?   // outside diameter (mm) when applicable
   115→  isVerified      Boolean  @default(false)
   116→  verifiedById    String?
   117→  verifiedAt      DateTime?
   118→  notes           String?
   119→  createdAt       DateTime @default(now())
   120→  updatedAt       DateTime @updatedAt
   121→
   122→  verifiedBy      User?    @relation("PipeStandardVerifiedBy", fields: [verifiedById], references: [id], onDelete: SetNull)
   123→  measurements    ThicknessMeasurement[]
   124→
   125→  @@unique([standard, edition, nominalSizeInch, schedule])
   126→  @@index([standard, nominalSizeInch])
   127→}

   392→// =====================================================
   393→// THICKNESS MEASUREMENT ROW
   394→// =====================================================
   395→// One row per measurement point. The first two columns (Dia mm (inch),
   396→// Pipe Schedule) are stored as structured values; the third column
   397→// (Nominal Thickness) is calculated from PipeStandard and may be
   398→// overridden only with a recorded reason. The four clock-position
   399→// measurements support an explicit status (Measured | NotMeasured |
   400→// NotApplicable) so missing data is never converted to zero.
   401→
   402→model ThicknessMeasurement {
   403→  id                    String   @id @default(cuid())
   404→  pipelineInspectionId String
   405→  pointNo               String   // free text, e.g. "P-1"
   406→  sortOrder             Int      @default(0)
   407→
   408→  // ─── Multi-table grouping (additive, default "default") ───
   409→  // When a Thickness report contains multiple tables, each row belongs to
   410→  // exactly one table group. Existing rows (and any new row in a
   411→  // single-table report) get "default" — so the existing single-table
   412→  // workflow keeps working unchanged. The list of table groups itself is
   413→  // stored as JSON on the parent PipelineInspection.thicknessTablesConfig.
   414→  tableGroup            String   @default("default")
   415→
   416→  // First two fixed columns
   417→  pipeSizeInch          String   // e.g. "2", "1/2", "6" (matches PipeStandard.nominalSizeInch)
   418→  pipeSchedule          String   // e.g. "SCH 40", "STD"
   419→
   420→  // Third column (calculated from standards; override captured separately)
   421→  nominalThicknessMm    Float?   // snapshot from PipeStandard at row creation/update
   422→  nominalOverride       Float?   // authorized override value (with reason at report level)
   423→  standardId            String?  // FK to PipeStandard used to compute nominalThicknessMm
   424→
   425→  // Fourth column
   426→  componentTypeId        String?
   427→  componentTypeCode      String   // snapshot code (PIPE | ELBOW | TEE | REDUCER | FLANGE | OTHER)
   428→
   429→  // Columns 5-9: clock-position measurements with explicit status
   430→  northMm               Float?
   431→  northStatus            String   @default("NotMeasured") // Measured | NotMeasured | NotApplicable
   432→  eastMm                 Float?
   433→  eastStatus             String   @default("NotMeasured")
   434→  southMm               Float?
   435→  southStatus            String   @default("NotMeasured")
   436→  westMm                 Float?
   437→  westStatus             String   @default("NotMeasured")
   438→
   439→  // Column 10: observed minimum (calculated; null if no valid measurements)
   440→  observedMinimumMm     Float?
   441→  observedMinimumOverride Float? // authorized override with reason
   442→
   443→  // Column 11: findings / comments
   444→  findingsComments      String?
   445→  // Custom column values stored as JSON: {"col_id_1": "5.2", "col_id_2": "N/A"}
   446→  // Each key corresponds to a custom column defined on the parent PipelineInspection.
   447→  customColumnValues    String?
   448→
   449→  createdAt             DateTime @default(now())
   450→  updatedAt             DateTime @updatedAt
   451→
   452→  pipelineInspection    PipelineInspection @relation(fields: [pipelineInspectionId], references: [id], onDelete: Cascade)
   453→  componentType         ComponentType?     @relation(fields: [componentTypeId], references: [id], onDelete: SetNull)
   454→  standard              PipeStandard?      @relation(fields: [standardId], references: [id], onDelete: SetNull)
   455→
   456→  @@index([pipelineInspectionId])
   457→  @@index([pipeSizeInch, pipeSchedule])
   458→}

   500→// Every save of an Approved report creates a new revision snapshot.

   503→model ReportRevision {

   522→  approvedByName   String?
   523→  createdAt         DateTime @default(now())
   524→
   525→  pipelineInspection PipelineInspection @relation(fields: [pipelineInspectionId], references: [id], onDelete: Cascade)
   526→  revisedBy          User?             @relation("ReportRevisionRevisedBy", fields: [revisedById], references: [id], onDelete: SetNull)
   527→  approvedBy         User?             @relation("ReportRevisionApprovedBy", fields: [approvedById], references: [id], onDelete: SetNull)
   528→
   529→  @@index([pipelineInspectionId])
   530→  @@index([status])
   531→}
