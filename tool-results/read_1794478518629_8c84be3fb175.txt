   182→// =====================================================
   183→// PIPE STANDARD REFERENCE DATA (ASME B36.10M / B36.19M)
   184→// =====================================================
   185→// Wall thickness values seeded from the publicly published ASME
   186→// reference tables for the most commonly used nominal pipe sizes and
   187→// schedules. These values MUST be re-verified against the licensed /
   188→// official ASME standard edition by an authorized administrator before
   189→// production use. An admin can edit, add, or delete entries, and every
   190→// edit is recorded with verifiedBy + verifiedAt + audit log entry.
   191→
   192→model PipeStandard {
   193→  id              String   @id @default(cuid())
   194→  standard        String   // "ASME B36.10M" | "ASME B36.19M"
   195→  edition         String   // e.g. "2018"
   196→  nominalSizeInch String   // e.g. "2" or "1/2"
   197→  nominalSizeMm   Float   // e.g. 60.3
   198→  dn              String   // e.g. "DN50"
   199→  schedule        String   // e.g. "SCH 40", "STD", "XS", "XXS", "SCH 40S"
   200→  wallThicknessMm Float   // nominal wall thickness in mm
   201→  oD_mm           Float?   // outside diameter (mm) when applicable
   202→  isVerified      Boolean  @default(false)
   203→  verifiedById    String?
   204→  verifiedAt      DateTime?
   205→  notes           String?
   206→  createdAt       DateTime @default(now())
   207→  updatedAt       DateTime @updatedAt
   208→
   209→  verifiedBy      User?    @relation("PipeStandardVerifiedBy", fields: [verifiedById], references: [id], onDelete: SetNull)
   210→  measurements    ThicknessMeasurement[]
   211→
   212→  @@unique([standard, edition, nominalSizeInch, schedule])
   213→  @@index([standard, nominalSizeInch])
   214→}

   479→// =====================================================
   480→// THICKNESS MEASUREMENT ROW
   481→// =====================================================
   482→// One row per measurement point. The first two columns (Dia mm (inch),
   483→// Pipe Schedule) are stored as structured values; the third column
   484→// (Nominal Thickness) is calculated from PipeStandard and may be
   485→// overridden only with a recorded reason. The four clock-position
   486→// measurements support an explicit status (Measured | NotMeasured |
   487→// NotApplicable) so missing data is never converted to zero.
   488→
   489→model ThicknessMeasurement {
   490→  id                    String   @id @default(cuid())
   491→  pipelineInspectionId String
   492→  pointNo               String   // free text, e.g. "P-1"
   493→  sortOrder             Int      @default(0)
   494→
   495→  // ─── Multi-table grouping (additive, default "default") ───
   496→  // When a Thickness report contains multiple tables, each row belongs to
   497→  // exactly one table group. Existing rows (and any new row in a
   498→  // single-table report) get "default" — so the existing single-table
   499→  // workflow keeps working unchanged. The list of table groups itself is
   500→  // stored as JSON on the parent PipelineInspection.thicknessTablesConfig.
   501→  tableGroup            String   @default("default")
   502→
   503→  // First two fixed columns
   504→  pipeSizeInch          String   // e.g. "2", "1/2", "6" (matches PipeStandard.nominalSizeInch)
   505→  pipeSchedule          String   // e.g. "SCH 40", "STD"
   506→
   507→  // Third column (calculated from standards; override captured separately)
   508→  nominalThicknessMm    Float?   // snapshot from PipeStandard at row creation/update
   509→  nominalOverride       Float?   // authorized override value (with reason at report level)
   510→  standardId            String?  // FK to PipeStandard used to compute nominalThicknessMm
   511→
   512→  // Fourth column
   513→  componentTypeId        String?
   514→  componentTypeCode      String   // snapshot code (PIPE | ELBOW | TEE | REDUCER | FLANGE | OTHER)
   515→
   516→  // Columns 5-9: clock-position measurements with explicit status
   517→  northMm               Float?
   518→  northStatus            String   @default("NotMeasured") // Measured | NotMeasured | NotApplicable
   519→  eastMm                 Float?
   520→  eastStatus             String   @default("NotMeasured")
   521→  southMm               Float?
   522→  southStatus            String   @default("NotMeasured")
   523→  westMm                 Float?
   524→  westStatus             String   @default("NotMeasured")
   525→
   526→  // Column 10: observed minimum (calculated; null if no valid measurements)
   527→  observedMinimumMm     Float?
   528→  observedMinimumOverride Float? // authorized override with reason
   529→
   530→  // Column 11: findings / comments
   531→  findingsComments      String?
   532→  // Custom column values stored as JSON: {"col_id_1": "5.2", "col_id_2": "N/A"}
   533→  // Each key corresponds to a custom column defined on the parent PipelineInspection.
   534→  customColumnValues    String?
   535→
   536→  createdAt             DateTime @default(now())
   537→  updatedAt             DateTime @updatedAt
   538→
   539→  pipelineInspection    PipelineInspection @relation(fields: [pipelineInspectionId], references: [id], onDelete: Cascade)
   540→  componentType         ComponentType?     @relation(fields: [componentTypeId], references: [id], onDelete: SetNull)
   541→  standard              PipeStandard?      @relation(fields: [standardId], references: [id], onDelete: SetNull)
   542→
   543→  @@index([pipelineInspectionId])
   544→  @@index([pipeSizeInch, pipeSchedule])
   545→}
